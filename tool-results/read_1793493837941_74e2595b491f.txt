    16→function formatRelativeTime(ts) {
    17→  if (!ts) return 'استفاده‌نشده';
    18→  const diffSec = Math.max(0, Math.floor((Date.now() - ts) / 1000));
    19→  if (diffSec < 45) return 'هم‌اکنون';
    20→  const mins = Math.floor(diffSec / 60);
    21→  if (mins < 60) return `${toFa(mins)} دقیقه پیش`;
    22→  const hours = Math.floor(mins / 60);
    23→  if (hours < 24) return `${toFa(hours)} ساعت پیش`;
    24→  const days = Math.floor(hours / 24);
    25→  return `${toFa(days)} روز پیش`;
    26→}

    28→function sendMsg(payload) {
    29→  return new Promise((resolve) => {
    30→    chrome.runtime.sendMessage(payload, (res) => {
    31→      if (chrome.runtime.lastError) {
    32→        resolve({ ok: false, error: chrome.runtime.lastError.message });
    33→        return;
    34→      }
    35→      resolve(res || { ok: false, error: 'پاسخی دریافت نشد.' });
    36→    });
    37→  });
    38→}

   208→async function loadState() {
   209→  const [tab] = await chrome.tabs.query({ active: true, currentWindow: true });
   210→
   211→  const [res, mRes] = await Promise.all([
   212→    sendMsg({ type: 'GET_SITE_STATE', tabId: tab?.id, url: tab?.url }),
   213→    sendMsg({ type: 'MAIL_GET_STATE' })
   214→  ]);
   215→
   216→  if (mRes?.ok) {
   217→    mailState.activeMailboxId = mRes.activeMailboxId;
   218→    mailState.mailboxes = mRes.mailboxes || [];
   219→    await loadMailUiPref();
   220→    renderMailSections();
   221→  }
   222→
   223→  if (!res.ok || !res.supported) {
   224→    state.supported = false;
   225→    state.tabId = tab?.id || null;
   226→    els.domainBadge.textContent = 'صفحه داخلی مرورگر';
   227→    els.countBadge.textContent = '۰ اکانت';
   228→    els.isoCountBadge.classList.add('hidden');
   229→    els.unsupportedState.classList.remove('hidden');
   230→    els.mainContent.classList.add('hidden');
   231→    return;
   232→  }
   233→
   234→  state = {
   235→    ...state,
   236→    tabId: tab?.id || res.tabId,
   237→    url: tab?.url || res.url,
   238→    supported: true,
   239→    siteKey: res.siteKey,
   240→    hostname: res.hostname,
   241→    isGoogleUnified: Boolean(res.isGoogleUnified),
   242→    activeAccountId: res.activeAccountId,
   243→    isIsolatedTab: Boolean(res.isIsolatedTab),
   244→    isolatedAccountId: res.isolatedAccountId || null,
   245→    openIsolatedTabs: res.openIsolatedTabs || [],
   246→    accounts: res.accounts || [],
   247→    settings: res.settings || state.settings
   248→  };
   249→
   250→  els.unsupportedState.classList.add('hidden');
   251→  els.mainContent.classList.remove('hidden');
   252→
   253→  const currentVal = els.accountNameInput.value.trim();
   254→  if (!currentVal || /^اکانت\s+[۰-۹0-9]+$/.test(currentVal)) {
   255→    els.accountNameInput.value = `اکانت ${toFa(state.accounts.length + 1)}`;
   256→  }
   257→
   258→  renderUI();
   259→}

   316→  els.emptyAccountsState.classList.toggle('hidden', total > 0);

   881→function htmlToCleanText(htmlArr, fallbackText) {
   882→  if (fallbackText && fallbackText.trim()) return fallbackText.trim();
   883→  if (!Array.isArray(htmlArr) || !htmlArr.length) return '(پیام بدون متن)';
   884→  try {
   885→    const doc = new DOMParser().parseFromString(htmlArr.join('\n'), 'text/html');
   886→    doc.querySelectorAll('script, style, noscript, iframe, object').forEach((el) => el.remove());
   887→    return (doc.body?.innerText || doc.body?.textContent || '').trim() || '(پیام بدون متن)';
   888→  } catch {
   889→    return '(خطا در نمایش متن پیام)';
   890→  }
   891→}
