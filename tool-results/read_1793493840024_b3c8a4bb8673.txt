    16→    16→function formatRelativeTime(ts) {
    17→    17→  if (!ts) return 'استفاده‌نشده';
    18→    18→  const diffSec = Math.max(0, Math.floor((Date.now() - ts) / 1000));
    19→    19→  if (diffSec < 45) return 'هم‌اکنون';
    20→    20→  const mins = Math.floor(diffSec / 60);
    21→    21→  if (mins < 60) return `${toFa(mins)} دقیقه پیش`;
    22→    22→  const hours = Math.floor(mins / 60);
    23→    23→  if (hours < 24) return `${toFa(hours)} ساعت پیش`;
    24→    24→  const days = Math.floor(hours / 24);
    25→    25→  return `${toFa(days)} روز پیش`;
    26→    26→}

   208→   208→async function loadState() {
   209→   209→  const [tab] = await chrome.tabs.query({ active: true, currentWindow: true });
   210→   210→
   211→   211→  const [res, mRes] = await Promise.all([
   212→   212→    sendMsg({ type: 'GET_SITE_STATE', tabId: tab?.id, url: tab?.url }),
   213→   213→    sendMsg({ type: 'MAIL_GET_STATE' })
   214→   214→  ]);
   215→   215→
   216→   216→  if (mRes?.ok) {
   217→   217→    mailState.activeMailboxId = mRes.activeMailboxId;
   218→   218→    mailState.mailboxes = mRes.mailboxes || [];
   219→   219→    await loadMailUiPref();
   220→   220→    renderMailSections();
   221→   221→  }
   222→   222→
   223→   223→  if (!res.ok || !res.supported) {
   224→   224→    state.supported = false;
   225→   225→    state.tabId = tab?.id || null;
   226→   226→    els.domainBadge.textContent = 'صفحه داخلی مرورگر';
   227→   227→    els.countBadge.textContent = '۰ اکانت';
   228→   228→    els.isoCountBadge.classList.add('hidden');
   229→   229→    els.unsupportedState.classList.remove('hidden');
   230→   230→    els.mainContent.classList.add('hidden');
   231→   231→    return;
   232→   232→  }
   233→   233→
   234→   234→  state = {
   235→   235→    ...state,
   236→   236→    tabId: tab?.id || res.tabId,
   237→   237→    url: tab?.url || res.url,
   238→   238→    supported: true,
   239→   239→    siteKey: res.siteKey,
   240→   240→    hostname: res.hostname,
   241→   241→    isGoogleUnified: Boolean(res.isGoogleUnified),
   242→   242→    activeAccountId: res.activeAccountId,
   243→   243→    isIsolatedTab: Boolean(res.isIsolatedTab),
   244→   244→    isolatedAccountId: res.isolatedAccountId || null,
   245→   245→    openIsolatedTabs: res.openIsolatedTabs || [],
   246→   246→    accounts: res.accounts || [],
   247→   247→    settings: res.settings || state.settings
   248→   248→  };
   249→   249→
   250→   250→  els.unsupportedState.classList.add('hidden');
   251→   251→  els.mainContent.classList.remove('hidden');

   253→   253→  const currentVal = els.accountNameInput.value.trim();
   254→   254→  if (!currentVal || /^اکانت\s+[۰-۹0-9]+$/.test(currentVal)) {
   255→   255→    els.accountNameInput.value = `اکانت ${toFa(state.accounts.length + 1)}`;
   256→   256→  }
   257→   257→
   258→   258→  renderUI();
   259→   259→}

   881→   881→function htmlToCleanText(htmlArr, fallbackText) {
   882→   882→  if (fallbackText && fallbackText.trim()) return fallbackText.trim();
   883→   883→  if (!Array.isArray(htmlArr) || !htmlArr.length) return '(پیام بدون متن)';
   884→   884→  try {
   885→   885→    const doc = new DOMParser().parseFromString(htmlArr.join('\n'), 'text/html');
