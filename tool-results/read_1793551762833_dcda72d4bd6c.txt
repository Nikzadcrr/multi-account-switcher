    59→async function ensureSessionHydrated() {
    60→  if (sessionHydrated) return;
    61→  sessionHydrated = true;
    62→  try {
    63→    if (chrome.storage?.session) {
    64→      const data = await chrome.storage.session.get(['isolatedTabsObj', 'nextDnrRuleId']);
    65→      if (data.isolatedTabsObj && typeof data.isolatedTabsObj === 'object') {
    66→        for (const [k, v] of Object.entries(data.isolatedTabsObj)) {
    67→          isolatedTabs.set(Number(k), v);
    68→        }
    69→      }
    70→      if (typeof data.nextDnrRuleId === 'number' && data.nextDnrRuleId > nextDnrRuleId) {
    71→        nextDnrRuleId = data.nextDnrRuleId;
    72→      }
    73→    }
    74→    if (chrome.declarativeNetRequest?.getSessionRules) {
    75→      const existingRules = await chrome.declarativeNetRequest.getSessionRules();
    76→      for (const r of existingRules) {
    77→        if (r.id >= nextDnrRuleId) {
    78→          nextDnrRuleId = r.id + 1;
    79→        }
    80→      }
    81→    }
    82→  } catch {}
    83→}

    86→  const id = nextDnrRuleId++;
    87→  if (nextDnrRuleId > 2000000000) nextDnrRuleId = 1000;
    88→  return id;

   113→function isIpOrLocalhost(hostname) {
   114→  if (hostname === 'localhost' || hostname.endsWith('.localhost')) return true;
   115→  if (/^\d{1,3}(\.\d{1,3}){3}$/.test(hostname)) return true;
   116→  if (hostname.startsWith('[') && hostname.endsWith(']')) return true;
   117→  return false;
   118→}

   120→function extractBaseDomain(hostname) {
   121→  const clean = hostname.toLowerCase().replace(/\.$/, '');
   122→  if (isIpOrLocalhost(clean)) return clean;
   123→
   124→  const parts = clean.split('.');
   125→  if (parts.length <= 2) return clean;
   126→
   127→  const lastTwo = parts.slice(-2).join('.');
   128→  if (TWO_PART_TLDS.has(lastTwo) && parts.length >= 3) {
   129→    return parts.slice(-3).join('.');
   130→  }
   131→  return lastTwo;
   132→}

   134→function getSiteInfo(urlStr) {
   135→  if (!urlStr) return { supported: false };
   136→  try {
   137→    const url = new URL(urlStr);
   138→    if (url.protocol !== 'http:' && url.protocol !== 'https:') {
   139→      return { supported: false };
   140→    }
   141→    if (
   142→      url.hostname === 'chromewebstore.google.com' ||
   143→      url.hostname === 'chrome.google.com'
   144→    ) {
   145→      return { supported: false, reason: 'restricted_store' };
   146→    }
   147→
   148→    const baseDomain = extractBaseDomain(url.hostname);
   149→    const cookieDomains = LINKED_AUTH_DOMAINS[baseDomain] || [baseDomain];
   150→
   151→    return {
   152→      supported: true,
   153→      url: urlStr,
   154→      origin: url.origin,
   155→      hostname: url.hostname,
   156→      siteKey: baseDomain,
   157→      baseDomain,
   158→      cookieDomains,
   159→      isGoogleUnified: baseDomain === 'google.com'
   160→    };
   161→  } catch {
   162→    return { supported: false };
   163→  }
   164→}

   257→async function captureCookies(cookieDomains, storeId) {
   258→  const map = new Map();
   259→
   260→  for (const domain of cookieDomains) {
   261→    const baseQuery = { domain };
   262→    if (storeId) baseQuery.storeId = storeId;
   263→
   264→    try {
   265→      const list = await chrome.cookies.getAll(baseQuery);
   266→      for (const c of list) {
   267→        const key = `${c.name}\0${c.domain}\0${c.path}\0${c.partitionKey?.topLevelSite || ''}`;
   268→        map.set(key, serializeCookie(c));
   269→      }
   270→    } catch {}
   271→
   272→    try {
   273→      const partList = await chrome.cookies.getAll({ ...baseQuery, partitionKey: {} });
   274→      for (const c of partList) {
   275→        const key = `${c.name}\0${c.domain}\0${c.path}\0${c.partitionKey?.topLevelSite || ''}`;
   276→        map.set(key, serializeCookie(c));
   277→      }
   278→    } catch {}
   279→  }
   280→
   281→  return Array.from(map.values());
   282→}

   284→function serializeCookie(c) {
   285→  const obj = {
   286→    name: c.name,
   287→    value: c.value,
   288→    domain: c.domain,
   289→    hostOnly: Boolean(c.hostOnly),
   290→    path: c.path || '/',
   291→    secure: Boolean(c.secure),
   292→    httpOnly: Boolean(c.httpOnly),
   293→    sameSite: c.sameSite || 'unspecified',
   294→    session: Boolean(c.session)
   295→  };
   296→  if (!c.session && typeof c.expirationDate === 'number') {
   297→    obj.expirationDate = c.expirationDate;
   298→  }
   299→  if (c.partitionKey) {
   300→    obj.partitionKey = c.partitionKey;
   301→  }
   302→  return obj;
   303→}

   305→function buildCookieUrl(c) {
   306→  const cleanHost = (c.domain || '').startsWith('.') ? c.domain.slice(1) : c.domain;
   307→  const isHostPrefix = (c.name || '').startsWith('__Host-');
   308→  const isSecurePrefix = (c.name || '').startsWith('__Secure-') || isHostPrefix;
   309→  const secure = Boolean(c.secure || isSecurePrefix || c.sameSite === 'no_restriction');
   310→  const path = isHostPrefix ? '/' : (c.path || '/');
   311→  return `${secure ? 'https://' : 'http://'}${cleanHost}${path}`;
   312→}

   314→async function clearCookies(cookieDomains, storeId) {
   315→  const existing = await captureCookies(cookieDomains, storeId);
   316→  await Promise.allSettled(
   317→    existing.map(async (c) => {
   318→      const url = buildCookieUrl(c);
   319→      const details = { url, name: c.name };
   320→      if (storeId) details.storeId = storeId;
   321→      if (c.partitionKey) details.partitionKey = c.partitionKey;
   322→      try {
   323→        await chrome.cookies.remove(details);
   324→      } catch {
   325→        const fallback = { url, name: c.name };
   326→        if (storeId) fallback.storeId = storeId;
   327→        await chrome.cookies.remove(fallback).catch(() => {});
   328→      }
   329→    })
   330→  );
   331→}

   333→async function restoreCookies(savedCookies, storeId) {
   334→  if (!Array.isArray(savedCookies) || savedCookies.length === 0) return;
   335→
   336→  const nowSec = Math.floor(Date.now() / 1000);
   337→
   338→  await Promise.allSettled(
   339→    savedCookies.map(async (c) => {
   340→      const isHostPrefix = (c.name || '').startsWith('__Host-');
   341→      const isSecurePrefix = (c.name || '').startsWith('__Secure-') || isHostPrefix;
   342→      const secure = Boolean(c.secure || isSecurePrefix || c.sameSite === 'no_restriction');
   343→      const path = isHostPrefix ? '/' : (c.path || '/');
   344→      const cleanHost = (c.domain || '').startsWith('.') ? c.domain.slice(1) : c.domain;
   345→      if (!cleanHost) return;
   346→
   347→      const url = `${secure ? 'https://' : 'http://'}${cleanHost}${path}`;
   348→      const details = {
   349→        url,
   350→        name: c.name,
   351→        value: c.value ?? '',
   352→        path,
   353→        secure,
   354→        httpOnly: Boolean(c.httpOnly)
   355→      };
   356→
   357→      if (!c.hostOnly && !isHostPrefix && c.domain) {
   358→        details.domain = c.domain;
   359→      }
   360→      if (c.sameSite && c.sameSite !== 'unspecified') {
   361→        details.sameSite = c.sameSite;
   362→      }
   363→      if (!c.session && typeof c.expirationDate === 'number') {
   364→        details.expirationDate =
   365→          c.expirationDate > nowSec ? c.expirationDate : nowSec + 86400 * 14;
   366→      }
   367→      if (storeId) {
   368→        details.storeId = storeId;
   369→      }
   370→      if (c.partitionKey) {
   371→        details.partitionKey = c.partitionKey;
   372→      }
   373→
   374→      try {
   375→        await chrome.cookies.set(details);
   376→      } catch {
   377→        if (details.partitionKey) {
   378→          delete details.partitionKey;
   379→          await chrome.cookies.set(details).catch(() => {});
   380→        }
   381→      }
   382→    })
   383→  );
   384→}

   390→function buildCookieHeaderForDomain(savedCookies, domain) {
   391→  const pairs = new Map();
   392→  for (const c of savedCookies || []) {
   393→    if (!c || !c.name) continue;
   394→    const clean = (c.domain || '').replace(/^\./, '').toLowerCase();
   395→    if (clean === domain || clean.endsWith('.' + domain)) {
   396→      // Keep first or root-path cookie
   397→      if (!pairs.has(c.name) || c.path === '/') {
   398→        pairs.set(c.name, c.value ?? '');
   399→      }
   400→    }
   401→  }
   402→  const list = [];
   403→  for (const [k, v] of pairs.entries()) {
   404→    list.push(`${k}=${v}`);
   405→  }
   406→  return list.join('; ');
   407→}

   409→function buildDnrCookieRulesForTab(tabId, cookieDomains, savedCookies) {
   410→  const rules = [];
   411→  for (const domain of cookieDomains) {
   412→    const cookieStr = buildCookieHeaderForDomain(savedCookies, domain);
   413→    const reqHeaderOp = cookieStr
   414→      ? { header: 'cookie', operation: 'set', value: cookieStr }
   415→      : { header: 'cookie', operation: 'remove' };
   416→
   417→    const condition = {
   418→      tabIds: [tabId],
   419→      resourceTypes: DNR_RESOURCE_TYPES
   420→    };
   421→    if (isIpOrLocalhost(domain)) {
   422→      condition.urlFilter = `||${domain}`;
   423→    } else {
   424→      condition.requestDomains = [domain];
   425→    }
   426→
   427→    rules.push({
   428→      id: allocRuleId(),
   429→      priority: 100,
   430→      action: {
   431→        type: 'modifyHeaders',
   432→        requestHeaders: [reqHeaderOp],
   433→        responseHeaders: [{ header: 'set-cookie', operation: 'remove' }]
   434→      },
   435→      condition
   436→    });
   437→  }
   438→  return rules;
   439→}

   441→async function refreshIsolatedTabDnrRules(tabId, account) {
   442→  await ensureSessionHydrated();
   443→  const iso = isolatedTabs.get(tabId);
   444→  if (!iso || !chrome.declarativeNetRequest) return;
   445→
   446→  const oldIds = iso.ruleIds || [];
   447→  const newRules = buildDnrCookieRulesForTab(
   448→    tabId,
   449→    iso.cookieDomains,
   450→    account.cookies || []
   451→  );
   452→  iso.ruleIds = newRules.map((r) => r.id);
   453→
   454→  await chrome.declarativeNetRequest.updateSessionRules({
   455→    removeRuleIds: oldIds,
   456→    addRules: newRules
   457→  });
   458→  await persistIsolatedSession();
   459→}

   470→      .catch(() => {});

   475→function waitForTabComplete(tabId, timeoutMs = 10000) {
   476→  return new Promise((resolve) => {
   477→    let done = false;
   478→    const finish = () => {
   479→      if (done) return;
   480→      done = true;
   481→      chrome.tabs.onUpdated.removeListener(listener);
   482→      clearTimeout(timer);
   483→      resolve();
   484→    };
   485→    const listener = (tid, info) => {
   486→      if (tid === tabId && info.status === 'complete') {
   487→        finish();
   488→      }
   489→    };
   490→    const timer = setTimeout(finish, timeoutMs);
   491→    chrome.tabs.onUpdated.addListener(listener);
   492→  });
   493→}

   495→async function openAccountInIsolatedTab(targetUrl, siteKey, accountId, makeActive = true) {
   496→  await ensureSessionHydrated();
   497→  const siteInfo = getSiteInfo(targetUrl);
   498→  if (!siteInfo.supported) {
   499→    return { ok: false, error: 'آدرس سایت برای تب مجزا معتبر نیست.' };
   500→  }
   501→
   502→  const record = await getDomainRecord(siteKey || siteInfo.siteKey);
   503→  const accIdx = record.accounts.findIndex((a) => a.id === accountId);
   504→  if (accIdx === -1) {
   505→    return { ok: false, error: 'اکانت مورد نظر پیدا نشد.' };
   506→  }
   507→  const account = record.accounts[accIdx];
   508→
   509→  // 1. Create blank tab first so we have tab.id BEFORE any network request happens
   510→  const tab = await chrome.tabs.create({ url: 'about:blank', active: Boolean(makeActive) });
   511→  const cookieRules = buildDnrCookieRulesForTab(
   512→    tab.id,
   513→    siteInfo.cookieDomains,
   514→    account.cookies || []
   515→  );
   516→  const bootCspRuleId = allocRuleId();
   517→
   518→  const bootRule = {
   519→    id: bootCspRuleId,
   520→    priority: 250,
   521→    action: {
   522→      type: 'modifyHeaders',
   523→      responseHeaders: [
   524→        {
   525→          header: 'content-security-policy',
   526→          operation: 'set',
   527→          value: "script-src 'none'"
   528→        }
   529→      ]
   530→    },
   531→    condition: {
   532→      tabIds: [tab.id],
   533→      resourceTypes: ['main_frame']
   534→    }
   535→  };
   536→
   537→  isolatedTabs.set(tab.id, {
   538→    tabId: tab.id,
   539→    siteKey: record.siteKey,
   540→    accountId: account.id,
   541→    cookieDomains: siteInfo.cookieDomains,
   542→    ruleIds: cookieRules.map((r) => r.id),
   543→    booting: true
   544→  });
   545→  await persistIsolatedSession();
   546→
   547→  await chrome.declarativeNetRequest.updateSessionRules({
   548→    addRules: [...cookieRules, bootRule]
   549→  });
   550→
   551→  // 2. Navigate once with page scripts blocked to seed per-tab sessionStorage on exact origin
   552→  const completePromise = waitForTabComplete(tab.id, 9000);
   553→  await chrome.tabs.update(tab.id, { url: targetUrl });
   554→  await completePromise;
   555→
   556→  // Prepare non-HttpOnly cookies map for document.cookie virtualization
   557→  const jsCookies = {};
   558→  for (const c of account.cookies || []) {
   559→    if (c && !c.httpOnly && c.name) {
   560→      jsCookies[c.name] = c.value ?? '';
   561→    }
   562→  }
   563→
   564→  const bootPayload = {
   565→    siteKey: record.siteKey,
   566→    accountId: account.id,
   567→    accountName: account.name,
   568→    shortBadge: toPersianDigits(accIdx + 1),
   569→    color: account.color || '#06b6d4',
   570→    localStorage: account.localStorage || {},
   571→    sessionStorage: account.sessionStorage || {},
   572→    jsCookies
   573→  };
   574→
   575→  await chrome.scripting
   576→    .executeScript({
   577→      target: { tabId: tab.id },
   578→      func: (payload) => {
   579→        try {
   580→          window.sessionStorage.setItem('__AI_SWITCHER_ISO__', JSON.stringify(payload));
   581→        } catch {}
   582→      },
   583→      args: [bootPayload]
   584→    })
   585→    .catch(() => {});
   586→
   587→  // 3. Remove temporary CSP script-blocker and reload tab for real execution!
   588→  await chrome.declarativeNetRequest
   589→    .updateSessionRules({ removeRuleIds: [bootCspRuleId] })
   590→    .catch(() => {});
   591→
   592→  const isoEntry = isolatedTabs.get(tab.id);
   593→  if (isoEntry) {
   594→    isoEntry.booting = false;
   595→    await persistIsolatedSession();
   596→  }
   597→
   598→  // Update usage stats
   599→  await withSiteLock(record.siteKey, async () => {
   600→    const freshRec = await getDomainRecord(record.siteKey);
   601→    const targetAcc = freshRec.accounts.find((a) => a.id === account.id);
   602→    if (targetAcc) {

   620→// Observe Set-Cookie headers on Isolated Tabs so rotated tokens stay fresh!
   621→const pendingCookieSaves = new Map();

   623→function parseSetCookieHeader(rawHeader, fallbackHost) {
   624→  if (!rawHeader || typeof rawHeader !== 'string') return null;
   625→  const parts = rawHeader.split(';').map((s) => s.trim());
   626→  const first = parts[0] || '';
   627→  const eq = first.indexOf('=');
   628→  if (eq <= 0) return null;
   629→
   630→  const name = first.slice(0, eq).trim();
   631→  const value = first.slice(eq + 1);
   632→  let domain = fallbackHost;
   633→  let hostOnly = true;
   634→  let path = '/';
   635→  let secure = true;
   636→  let httpOnly = false;
   637→  let expired = false;
   638→  let expirationDate;
   639→
   640→  for (let i = 1; i < parts.length; i++) {
   641→    const seg = parts[i];
   642→    const low = seg.toLowerCase();
   643→    if (low === 'httponly') {
   644→      httpOnly = true;
   645→    } else if (low === 'secure') {
   646→      secure = true;
   647→    } else if (low.startsWith('domain=')) {
   648→      const d = seg.slice(7).trim().toLowerCase();
   649→      if (d) {
   650→        domain = d.startsWith('.') ? d : '.' + d;
   651→        hostOnly = false;
   652→      }
   653→    } else if (low.startsWith('path=')) {
   654→      path = seg.slice(5).trim() || '/';
   655→    } else if (low.startsWith('max-age=')) {
   656→      const sec = Number(seg.slice(8).trim());
   657→      if (!Number.isNaN(sec)) {
   658→        if (sec <= 0) expired = true;
   659→        else expirationDate = Math.floor(Date.now() / 1000) + sec;
   660→      }
   661→    } else if (low.startsWith('expires=')) {
   662→      const ms = Date.parse(seg.slice(8).trim());
   663→      if (!Number.isNaN(ms)) {
   664→        if (ms <= Date.now()) expired = true;
   665→        else expirationDate = Math.floor(ms / 1000);
   666→      }
   667→    }
   668→  }
   669→
   670→  return {
   671→    name,
   672→    value,
   673→    domain,
   674→    hostOnly,
   675→    path,
   676→    secure,
   677→    httpOnly,
   678→    sameSite: 'lax',
   679→    session: !expirationDate,
   680→    ...(expirationDate ? { expirationDate } : {}),
   681→    expired
   682→  };
   683→}
   684→
   685→if (chrome.webRequest?.onHeadersReceived) {
   686→  chrome.webRequest.onHeadersReceived.addListener(
   687→    (details) => {
   688→      if (details.tabId < 0) return;
   689→      const iso = isolatedTabs.get(details.tabId);
   690→      if (!iso || iso.booting) return;
   691→
   692→      const setCookieHeaders = (details.responseHeaders || []).filter(
   693→        (h) => h.name && h.name.toLowerCase() === 'set-cookie' && h.value
   694→      );
   695→      if (setCookieHeaders.length === 0) return;
   696→
   697→      let reqHost = iso.siteKey;
   698→      try {
   699→        reqHost = new URL(details.url).hostname.toLowerCase();
   700→      } catch {}
   701→
   702→      const parsedList = setCookieHeaders
   703→        .map((h) => parseSetCookieHeader(h.value, reqHost))
   704→        .filter(Boolean);
   705→      if (parsedList.length === 0) return;
   706→
   707→      // Queue async update for this isolated account
   708→      const key = `${details.tabId}:${iso.accountId}`;
   709→      const existing = pendingCookieSaves.get(key) || [];
   710→      existing.push(...parsedList);
   711→      pendingCookieSaves.set(key, existing);
   712→
   713→      setTimeout(() => {
   714→        const batch = pendingCookieSaves.get(key);
   715→        if (!batch || batch.length === 0) return;
   716→        pendingCookieSaves.delete(key);
   717→
   718→        withSiteLock(iso.siteKey, async () => {
   719→          const rec = await getDomainRecord(iso.siteKey);
   720→          const acc = rec.accounts.find((a) => a.id === iso.accountId);
   721→          if (!acc) return;
   722→
   723→          const cookies = Array.isArray(acc.cookies) ? [...acc.cookies] : [];
   724→          for (const item of batch) {
   725→            const idx = cookies.findIndex(
   726→              (c) =>
   727→                c.name === item.name &&
   728→                c.domain.replace(/^\./, '') === item.domain.replace(/^\./, '')
   729→            );
   730→            if (item.expired) {
   731→              if (idx >= 0) cookies.splice(idx, 1);
   732→            } else {
   733→              const cleanObj = { ...item };
   734→              delete cleanObj.expired;
   735→              if (idx >= 0) cookies[idx] = { ...cookies[idx], ...cleanObj };
   736→              else cookies.push(cleanObj);
   737→            }
   738→          }
   739→
   740→          acc.cookies = cookies;
   741→          acc.updatedAt = Date.now();
   742→          await saveDomainRecord(iso.siteKey, rec);
   743→          await refreshIsolatedTabDnrRules(details.tabId, acc);
   744→        });
   745→      }, 250);
   746→    },
   747→    { urls: ['<all_urls>'] },
   748→    ['responseHeaders', 'extraHeaders']
   749→  );
   750→}

   756→async function captureTabStorage(tabId) {
   757→  if (!tabId) {
   758→    return { localStorage: {}, sessionStorage: {}, indexedDB: {} };
   759→  }
   760→  try {
   761→    const results = await chrome.scripting.executeScript({
   762→      target: { tabId },
   763→      func: async () => {
   764→        const MAX_ITEM_LEN = 512 * 1024;
   765→        const ls = {};
   766→        const ss = {};
   767→
   768→        try {
   769→          for (let i = 0; i < window.localStorage.length; i++) {
   770→            const k = window.localStorage.key(i);
   771→            if (k !== null && k !== '__AI_SWITCHER_ISO__') {
   772→              const v = window.localStorage.getItem(k);
   773→              if (typeof v === 'string' && v.length <= MAX_ITEM_LEN) {
   774→                ls[k] = v;
   775→              }
   776→            }
   777→          }
   778→        } catch {}
   779→
   780→        try {
   781→          for (let i = 0; i < window.sessionStorage.length; i++) {
   782→            const k = window.sessionStorage.key(i);
   783→            if (k !== null && k !== '__AI_SWITCHER_ISO__') {
   784→              const v = window.sessionStorage.getItem(k);
   785→              if (typeof v === 'string' && v.length <= MAX_ITEM_LEN) {
   786→                ss[k] = v;
   787→              }
   788→            }
   789→          }
   790→        } catch {}
   791→
   792→        const idbDump = {};
   793→        try {
   794→          if (window.indexedDB && typeof window.indexedDB.databases === 'function') {
   795→            const captureIdb = async () => {
   796→              const dbs = await window.indexedDB.databases();
   797→              const authPattern = /firebase|auth|session|token|supabase|clerk|keyval|user/i;
   798→              for (const info of dbs) {
   799→                if (!info.name || !authPattern.test(info.name)) continue;
   800→                await new Promise((resolve) => {
   801→                  let settled = false;
   802→                  const done = () => {
   803→                    if (!settled) {
   804→                      settled = true;
   805→                      resolve();
   806→                    }
   807→                  };
   808→                  const timer = setTimeout(done, 400);
   809→                  try {
   810→                    const req = window.indexedDB.open(info.name);
   811→                    req.onerror = () => {
   812→                      clearTimeout(timer);
   813→                      done();
   814→                    };
   815→                    req.onsuccess = () => {
   816→                      const db = req.result;
   817→                      const storeNames = Array.from(db.objectStoreNames || []);
   818→                      if (storeNames.length === 0) {
   819→                        db.close();
   820→                        clearTimeout(timer);
   821→                        done();
   822→                        return;
   823→                      }
   824→                      const dbRecord = {};
   825→                      try {
   826→                        const tx = db.transaction(storeNames, 'readonly');
   827→                        let pending = storeNames.length;
   828→                        for (const sName of storeNames) {
   829→                          const store = tx.objectStore(sName);
   830→                          const usesInlineKey = store.keyPath !== null;
   831→                          const valReq = store.getAll();
   832→                          const keyReq = store.getAllKeys();
   833→                          let vals = null;
   834→                          let keys = null;
   835→                          const checkStoreDone = () => {
   836→                            if (vals !== null && keys !== null) {
   837→                              try {
   838→                                const jsonSize = JSON.stringify(vals).length;
   839→                                if (jsonSize <= 256 * 1024) {
   840→                                  dbRecord[sName] = {
   841→                                    usesInlineKey,
   842→                                    entries: vals.map((v, idx) => ({
   843→                                      key: keys[idx],
   844→                                      value: v
   845→                                    }))
   846→                                  };
   847→                                }
   848→                              } catch {}
   849→                              pending--;
   850→                              if (pending <= 0) {
   851→                                idbDump[info.name] = dbRecord;
   852→                                db.close();
   853→                                clearTimeout(timer);
   854→                                done();
   855→                              }
   856→                            }
   857→                          };
   858→                          valReq.onsuccess = () => {
   859→                            vals = valReq.result || [];
   860→                            checkStoreDone();
   861→                          };
   862→                          keyReq.onsuccess = () => {
   863→                            keys = keyReq.result || [];
   864→                            checkStoreDone();
   865→                          };
   866→                          valReq.onerror = keyReq.onerror = () => {
   867→                            vals = [];
   868→                            keys = [];
   869→                            checkStoreDone();
   870→                          };
   871→                        }
   872→                      } catch {
   873→                        db.close();
   874→                        clearTimeout(timer);
   875→                        done();
   876→                      }
   877→                    };
   878→                  } catch {
   879→                    clearTimeout(timer);
   880→                    done();
   881→                  }
   882→                });
   883→              }
   884→            };
   885→            await Promise.race([
   886→              captureIdb(),
   887→              new Promise((r) => setTimeout(r, 700))
   888→            ]);
   889→          }
   890→        } catch {}
   891→
   892→        return { localStorage: ls, sessionStorage: ss, indexedDB: idbDump };
   893→      }
   894→    });
   895→
   896→    if (results && results[0] && results[0].result) {
   897→      return results[0].result;
   898→    }
   899→  } catch {}
   900→  return { localStorage: {}, sessionStorage: {}, indexedDB: {} };
   901→}

   903→async function applyTabStorageAndReload(tabId, storagePayload, includeStorage) {
   904→  if (!tabId) return;
   905→
   906→  if (includeStorage) {
   907→    try {
   908→      await chrome.scripting.executeScript({
   909→        target: { tabId },
   910→        func: async (payload) => {
   911→          try {
   912→            window.localStorage.clear();

   914→            for (const [k, v] of Object.entries(ls)) {
   915→              window.localStorage.setItem(k, v);

   918→
   919→          try {
   920→            window.sessionStorage.clear();

   922→            for (const [k, v] of Object.entries(ss)) {
   923→              window.sessionStorage.setItem(k, v);
   924→            }
   925→          } catch {}
   926→
   927→          try {
   928→            if (window.indexedDB && typeof window.indexedDB.databases === 'function') {
   929→              const syncIdb = async () => {
   930→                const targetIdb = (payload && payload.indexedDB) || {};
   931→                const existingDbs = await window.indexedDB.databases();
   932→                const authPattern = /firebase|auth|session|token|supabase|clerk|keyval|user/i;
   933→                const dbNames = new Set([
   934→                  ...Object.keys(targetIdb),
   935→                  ...existingDbs
   936→                    .map((d) => d.name)
   937→                    .filter((n) => n && authPattern.test(n))
   938→                ]);
   939→
   940→                for (const dbName of dbNames) {
   941→                  await new Promise((resolve) => {
   942→                    let settled = false;
   943→                    const done = () => {
   944→                      if (!settled) {
   945→                        settled = true;
   946→                        resolve();
   947→                      }
   948→                    };
   949→                    const timer = setTimeout(done, 400);
   950→                    try {
   951→                      const req = window.indexedDB.open(dbName);
   952→                      req.onerror = () => {
   953→                        clearTimeout(timer);
   954→                        done();
   955→                      };
   956→                      req.onsuccess = () => {
   957→                        const db = req.result;
   958→                        const storeNames = Array.from(db.objectStoreNames || []);
   959→                        if (storeNames.length === 0) {
   960→                          db.close();
   961→                          clearTimeout(timer);
   962→                          done();
   963→                          return;
   964→                        }
   965→                        try {
   966→                          const tx = db.transaction(storeNames, 'readwrite');
   967→                          const savedStores = targetIdb[dbName] || {};
   968→                          for (const sName of storeNames) {
   969→                            const store = tx.objectStore(sName);
   970→                            store.clear();
   971→                            const savedStore = savedStores[sName];
   972→                            if (savedStore && Array.isArray(savedStore.entries)) {
   973→                              for (const item of savedStore.entries) {
   974→                                try {
   975→                                  if (savedStore.usesInlineKey || store.keyPath !== null) {
   976→                                    store.put(item.value);
   977→                                  } else {
   978→                                    store.put(item.value, item.key);
   979→                                  }
   980→                                } catch {}
   981→                              }
   982→                            }
   983→                          }
   984→                          tx.oncomplete = tx.onerror = tx.onabort = () => {
   985→                            db.close();
   986→                            clearTimeout(timer);
   987→                            done();
   988→                          };
   989→                        } catch {
   990→                          db.close();
   991→                          clearTimeout(timer);
   992→                          done();
   993→                        }
   994→                      };
   995→                    } catch {
   996→                      clearTimeout(timer);
   997→                      done();
   998→                    }
   999→                  });
  1000→                }
  1001→              };
  1002→              await Promise.race([
  1003→                syncIdb(),
  1004→                new Promise((r) => setTimeout(r, 750))
  1005→              ]);
  1006→            }
  1007→          } catch {}
  1008→        },
  1009→        args: [storagePayload || { localStorage: {}, sessionStorage: {}, indexedDB: {} }]
  1010→      });
  1011→
  1012→      await chrome.scripting
  1013→        .executeScript({
  1014→          target: { tabId },
  1015→          world: 'MAIN',
  1016→          func: () => {
  1017→            try {
  1018→              Storage.prototype.setItem = function () {};
  1019→              Storage.prototype.removeItem = function () {};
  1020→              Storage.prototype.clear = function () {};
  1021→            } catch {}
  1022→          }
  1023→        })
  1024→        .catch(() => {});
  1025→    } catch {}
  1026→  }
  1027→
  1028→  await chrome.tabs.reload(tabId);
  1029→}

  1035→async function updateBadgeForTab(tabId, urlStr) {
  1036→  if (!tabId) return;

  1039→  const info = getSiteInfo(urlStr);

  1043→  }
  1044→

  1046→  if (!record.accounts.length) {
  1047→    await chrome.action.setBadgeText({ tabId, text: '' }).catch(() => {});
  1048→    return;
  1049→  }
  1050→

  1061→        .catch(() => {});

  1063→    }

  1103→async function maybeAutoSyncOutgoingAccount(record, siteInfo, tabId, storeId) {
  1104→  if (!record.settings.autoSync || !record.activeAccountId) return;
  1105→  // Do not overwrite shared activeAccountId from an isolated tab
  1106→  if (isolatedTabs.has(tabId)) return;
  1107→
  1108→  const outgoing = record.accounts.find((a) => a.id === record.activeAccountId);
  1109→  if (!outgoing) return;
  1110→
  1111→  const liveCookies = await captureCookies(siteInfo.cookieDomains, storeId);
  1112→  const prevCount = Array.isArray(outgoing.cookies) ? outgoing.cookies.length : 0;
  1113→
  1114→  if (liveCookies.length > 0 && liveCookies.length >= Math.floor(prevCount * 0.5)) {
  1115→    outgoing.cookies = liveCookies;
  1116→    if (record.settings.includeStorage) {
  1117→      const liveStorage = await captureTabStorage(tabId);
  1118→      const prevLsCount = Object.keys(outgoing.localStorage || {}).length;
  1119→      const liveLsCount = Object.keys(liveStorage.localStorage || {}).length;
  1120→      if (prevLsCount === 0 || liveLsCount >= Math.floor(prevLsCount * 0.5)) {
  1121→        outgoing.localStorage = liveStorage.localStorage;
  1122→        outgoing.sessionStorage = liveStorage.sessionStorage;
  1123→        outgoing.indexedDB = liveStorage.indexedDB;
  1124→      }
  1125→    }
  1126→    outgoing.updatedAt = Date.now();
  1127→  }
  1128→}

  1423→// ---------------- Instant Mail HD codec (استخراج‌شده از باندل اپ) ----------------
  1424→// ENC(x) = charSwap( reverse( base64( JSON(x) ) ) )

  1427→const HD_SWAP_PAIRS = [
  1428→  ['a', 'o'], ['s', 'q'], ['=', 'x'], ['5', 'b'], ['4', '*'], ['B', 'z']
  1429→];
  1430→
  1431→function hdSwapChars(str) {
  1432→  let out = '';
  1433→  for (const ch of str) {
  1434→    let c = ch;
  1435→    for (const [a, b] of HD_SWAP_PAIRS) {
  1436→      if (c === a) { c = b; break; }
  1437→      if (c === b) { c = a; break; }
  1438→    }
  1439→    out += c;
  1440→  }
  1441→  return out;
  1442→}

  1444→function hdEncode(obj) {
  1445→  const json = JSON.stringify(obj);
  1446→  const bytes = new TextEncoder().encode(json);
  1447→  let bin = '';
  1448→  for (let i = 0; i < bytes.length; i++) {
  1449→    bin += String.fromCharCode(bytes[i]);
  1450→  }
  1451→  const b64 = btoa(bin);
  1452→  return hdSwapChars(b64.split('').reverse().join(''));
  1453→}

  1459→  return new TextDecoder().decode(bytes);
