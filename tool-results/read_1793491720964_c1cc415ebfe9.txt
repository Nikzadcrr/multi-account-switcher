    59→async function ensureSessionHydrated() {
    60→  if (sessionHydrated) return;
    61→  sessionHydrated = true;
    62→  try {
    63→    if (chrome.storage?.session) {
    64→      const data = await chrome.storage.session.get(['isolatedTabsObj', 'nextDnrRuleId']);
    65→      if (data.isolatedTabsObj && typeof data.isolatedTabsObj === 'object') {
    66→        for (const [k, v] of Object.entries(data.isolatedTabsObj)) {
    67→          isolatedTabs.set(Number(k), v);
    68→        }
    69→      }
    70→      if (typeof data.nextDnrRuleId === 'number' && data.nextDnrRuleId > nextDnrRuleId) {
    71→        nextDnrRuleId = data.nextDnrRuleId;
    72→      }
    73→    }
    74→    if (chrome.declarativeNetRequest?.getSessionRules) {
    75→      const existingRules = await chrome.declarativeNetRequest.getSessionRules();
    76→      for (const r of existingRules) {
    77→        if (r.id >= nextDnrRuleId) {
    78→          nextDnrRuleId = r.id + 1;
    79→        }
    80→      }
    81→    }
    82→  } catch {}
    83→}

    86→  const id = nextDnrRuleId++;
    87→  if (nextDnrRuleId > 2000000000) nextDnrRuleId = 1000;
    88→  return id;

   113→function isIpOrLocalhost(hostname) {
   114→  if (hostname === 'localhost' || hostname.endsWith('.localhost')) return true;
   115→  if (/^\d{1,3}(\.\d{1,3}){3}$/.test(hostname)) return true;
   116→  if (hostname.startsWith('[') && hostname.endsWith(']')) return true;
   117→  return false;
   118→}

   120→function extractBaseDomain(hostname) {
   121→  const clean = hostname.toLowerCase().replace(/\.$/, '');
   122→  if (isIpOrLocalhost(clean)) return clean;
   123→
   124→  const parts = clean.split('.');
   125→  if (parts.length <= 2) return clean;
   126→
   127→  const lastTwo = parts.slice(-2).join('.');
   128→  if (TWO_PART_TLDS.has(lastTwo) && parts.length >= 3) {
   129→    return parts.slice(-3).join('.');
   130→  }
   131→  return lastTwo;
   132→}

   134→function getSiteInfo(urlStr) {
   135→  if (!urlStr) return { supported: false };
   136→  try {
   137→    const url = new URL(urlStr);
   138→    if (url.protocol !== 'http:' && url.protocol !== 'https:') {
   139→      return { supported: false };
   140→    }
   141→    if (
   142→      url.hostname === 'chromewebstore.google.com' ||
   143→      url.hostname === 'chrome.google.com'
   144→    ) {
   145→      return { supported: false, reason: 'restricted_store' };
   146→    }
   147→
   148→    const baseDomain = extractBaseDomain(url.hostname);
   149→    const cookieDomains = LINKED_AUTH_DOMAINS[baseDomain] || [baseDomain];
   150→
   151→    return {
   152→      supported: true,
   153→      url: urlStr,
   154→      origin: url.origin,
   155→      hostname: url.hostname,
   156→      siteKey: baseDomain,
   157→      baseDomain,
   158→      cookieDomains,
   159→      isGoogleUnified: baseDomain === 'google.com'
   160→    };
   161→  } catch {
   162→    return { supported: false };
   163→  }
   164→}

   255→async function captureCookies(cookieDomains, storeId) {
   256→  const map = new Map();
   257→
   258→  for (const domain of cookieDomains) {
   259→    const baseQuery = { domain };
   260→    if (storeId) baseQuery.storeId = storeId;
   261→
   262→    try {
   263→      const list = await chrome.cookies.getAll(baseQuery);
   264→      for (const c of list) {
   265→        const key = `${c.name}\0${c.domain}\0${c.path}\0${c.partitionKey?.topLevelSite || ''}`;
   266→        map.set(key, serializeCookie(c));
   267→      }
   268→    } catch {}
   269→
   270→    try {
   271→      const partList = await chrome.cookies.getAll({ ...baseQuery, partitionKey: {} });
   272→      for (const c of partList) {
   273→        const key = `${c.name}\0${c.domain}\0${c.path}\0${c.partitionKey?.topLevelSite || ''}`;
   274→        map.set(key, serializeCookie(c));
   275→      }
   276→    } catch {}
   277→  }
   278→
   279→  return Array.from(map.values());
   280→}

   282→function serializeCookie(c) {
   283→  const obj = {
   284→    name: c.name,
   285→    value: c.value,
   286→    domain: c.domain,
   287→    hostOnly: Boolean(c.hostOnly),
   288→    path: c.path || '/',
   289→    secure: Boolean(c.secure),
   290→    httpOnly: Boolean(c.httpOnly),
   291→    sameSite: c.sameSite || 'unspecified',
   292→    session: Boolean(c.session)
   293→  };
   294→  if (!c.session && typeof c.expirationDate === 'number') {
   295→    obj.expirationDate = c.expirationDate;
   296→  }
   297→  if (c.partitionKey) {
   298→    obj.partitionKey = c.partitionKey;
   299→  }
   300→  return obj;
   301→}

   303→function buildCookieUrl(c) {
   304→  const cleanHost = (c.domain || '').startsWith('.') ? c.domain.slice(1) : c.domain;
   305→  const isHostPrefix = (c.name || '').startsWith('__Host-');
   306→  const isSecurePrefix = (c.name || '').startsWith('__Secure-') || isHostPrefix;
   307→  const secure = Boolean(c.secure || isSecurePrefix || c.sameSite === 'no_restriction');
   308→  const path = isHostPrefix ? '/' : (c.path || '/');
   309→  return `${secure ? 'https://' : 'http://'}${cleanHost}${path}`;
   310→}

   312→async function clearCookies(cookieDomains, storeId) {
   313→  const existing = await captureCookies(cookieDomains, storeId);
   314→  await Promise.allSettled(
   315→    existing.map(async (c) => {
   316→      const url = buildCookieUrl(c);
   317→      const details = { url, name: c.name };
   318→      if (storeId) details.storeId = storeId;
   319→      if (c.partitionKey) details.partitionKey = c.partitionKey;
   320→      try {
   321→        await chrome.cookies.remove(details);
   322→      } catch {
   323→        const fallback = { url, name: c.name };
   324→        if (storeId) fallback.storeId = storeId;
   325→        await chrome.cookies.remove(fallback).catch(() => {});
   326→      }
   327→    })
   328→  );
   329→}

   331→async function restoreCookies(savedCookies, storeId) {
   332→  if (!Array.isArray(savedCookies) || savedCookies.length === 0) return;
   333→
   334→  const nowSec = Math.floor(Date.now() / 1000);
   335→
   336→  await Promise.allSettled(
   337→    savedCookies.map(async (c) => {
   338→      const isHostPrefix = (c.name || '').startsWith('__Host-');
   339→      const isSecurePrefix = (c.name || '').startsWith('__Secure-') || isHostPrefix;
   340→      const secure = Boolean(c.secure || isSecurePrefix || c.sameSite === 'no_restriction');
   341→      const path = isHostPrefix ? '/' : (c.path || '/');
   342→      const cleanHost = (c.domain || '').startsWith('.') ? c.domain.slice(1) : c.domain;
   343→      if (!cleanHost) return;
   344→
   345→      const url = `${secure ? 'https://' : 'http://'}${cleanHost}${path}`;
   346→      const details = {
   347→        url,
   348→        name: c.name,
   349→        value: c.value ?? '',
   350→        path,
   351→        secure,
   352→        httpOnly: Boolean(c.httpOnly)
   353→      };
   354→
   355→      if (!c.hostOnly && !isHostPrefix && c.domain) {
   356→        details.domain = c.domain;
   357→      }
   358→      if (c.sameSite && c.sameSite !== 'unspecified') {
   359→        details.sameSite = c.sameSite;
   360→      }
   361→      if (!c.session && typeof c.expirationDate === 'number') {
   362→        details.expirationDate =
   363→          c.expirationDate > nowSec ? c.expirationDate : nowSec + 86400 * 14;
   364→      }
   365→      if (storeId) {
   366→        details.storeId = storeId;
   367→      }
   368→      if (c.partitionKey) {
   369→        details.partitionKey = c.partitionKey;
   370→      }
   371→
   372→      try {
   373→        await chrome.cookies.set(details);
   374→      } catch {
   375→        if (details.partitionKey) {
   376→          delete details.partitionKey;
   377→          await chrome.cookies.set(details).catch(() => {});
   378→        }
   379→      }
   380→    })
   381→  );
   382→}

   388→function buildCookieHeaderForDomain(savedCookies, domain) {
   389→  const pairs = new Map();
   390→  for (const c of savedCookies || []) {
   391→    if (!c || !c.name) continue;
   392→    const clean = (c.domain || '').replace(/^\./, '').toLowerCase();
   393→    if (clean === domain || clean.endsWith('.' + domain)) {
   394→      // Keep first or root-path cookie
   395→      if (!pairs.has(c.name) || c.path === '/') {
   396→        pairs.set(c.name, c.value ?? '');
   397→      }
   398→    }
   399→  }
   400→  const list = [];
   401→  for (const [k, v] of pairs.entries()) {
   402→    list.push(`${k}=${v}`);
   403→  }
   404→  return list.join('; ');
   405→}

   407→function buildDnrCookieRulesForTab(tabId, cookieDomains, savedCookies) {
   408→  const rules = [];
   409→  for (const domain of cookieDomains) {
   410→    const cookieStr = buildCookieHeaderForDomain(savedCookies, domain);
   411→    const reqHeaderOp = cookieStr
   412→      ? { header: 'cookie', operation: 'set', value: cookieStr }
   413→      : { header: 'cookie', operation: 'remove' };
   414→
   415→    const condition = {
   416→      tabIds: [tabId],
   417→      resourceTypes: DNR_RESOURCE_TYPES
   418→    };
   419→    if (isIpOrLocalhost(domain)) {
   420→      condition.urlFilter = `||${domain}`;
   421→    } else {
   422→      condition.requestDomains = [domain];
   423→    }
   424→
   425→    rules.push({
   426→      id: allocRuleId(),
   427→      priority: 100,
   428→      action: {
   429→        type: 'modifyHeaders',
   430→        requestHeaders: [reqHeaderOp],
   431→        responseHeaders: [{ header: 'set-cookie', operation: 'remove' }]
   432→      },
   433→      condition
   434→    });
   435→  }
   436→  return rules;
   437→}

   439→async function refreshIsolatedTabDnrRules(tabId, account) {
   440→  await ensureSessionHydrated();
   441→  const iso = isolatedTabs.get(tabId);
   442→  if (!iso || !chrome.declarativeNetRequest) return;
   443→
   444→  const oldIds = iso.ruleIds || [];
   445→  const newRules = buildDnrCookieRulesForTab(
   446→    tabId,
   447→    iso.cookieDomains,
   448→    account.cookies || []
   449→  );
   450→  iso.ruleIds = newRules.map((r) => r.id);
   451→
   452→  await chrome.declarativeNetRequest.updateSessionRules({
   453→    removeRuleIds: oldIds,
   454→    addRules: newRules
   455→  });
   456→  await persistIsolatedSession();
   457→}

   473→function waitForTabComplete(tabId, timeoutMs = 10000) {
   474→  return new Promise((resolve) => {
   475→    let done = false;
   476→    const finish = () => {
   477→      if (done) return;
   478→      done = true;
   479→      chrome.tabs.onUpdated.removeListener(listener);
   480→      clearTimeout(timer);
   481→      resolve();
   482→    };
   483→    const listener = (tid, info) => {
   484→      if (tid === tabId && info.status === 'complete') {
   485→        finish();
   486→      }
   487→    };
   488→    const timer = setTimeout(finish, timeoutMs);
   489→    chrome.tabs.onUpdated.addListener(listener);
   490→  });
   491→}

   493→async function openAccountInIsolatedTab(targetUrl, siteKey, accountId, makeActive = true) {
   494→  await ensureSessionHydrated();
   495→  const siteInfo = getSiteInfo(targetUrl);
   496→  if (!siteInfo.supported) {
   497→    return { ok: false, error: 'آدرس سایت برای تب مجزا معتبر نیست.' };
   498→  }
   499→
   500→  const record = await getDomainRecord(siteKey || siteInfo.siteKey);
   501→  const accIdx = record.accounts.findIndex((a) => a.id === accountId);
   502→  if (accIdx === -1) {
   503→    return { ok: false, error: 'اکانت مورد نظر پیدا نشد.' };
   504→  }
   505→  const account = record.accounts[accIdx];
   506→
   507→  // 1. Create blank tab first so we have tab.id BEFORE any network request happens
   508→  const tab = await chrome.tabs.create({ url: 'about:blank', active: Boolean(makeActive) });
   509→  const cookieRules = buildDnrCookieRulesForTab(
   510→    tab.id,
   511→    siteInfo.cookieDomains,
   512→    account.cookies || []
   513→  );
   514→  const bootCspRuleId = allocRuleId();
   515→
   516→  const bootRule = {
   517→    id: bootCspRuleId,
   518→    priority: 250,
   519→    action: {
   520→      type: 'modifyHeaders',
   521→      responseHeaders: [
   522→        {
   523→          header: 'content-security-policy',
   524→          operation: 'set',
   525→          value: "script-src 'none'"
   526→        }
   527→      ]
   528→    },
   529→    condition: {
   530→      tabIds: [tab.id],
   531→      resourceTypes: ['main_frame']
   532→    }
   533→  };
   534→
   535→  isolatedTabs.set(tab.id, {
   536→    tabId: tab.id,
   537→    siteKey: record.siteKey,
   538→    accountId: account.id,
   539→    cookieDomains: siteInfo.cookieDomains,
   540→    ruleIds: cookieRules.map((r) => r.id),
   541→    booting: true
   542→  });
   543→  await persistIsolatedSession();
   544→
   545→  await chrome.declarativeNetRequest.updateSessionRules({
   546→    addRules: [...cookieRules, bootRule]
   547→  });
   548→
   549→  // 2. Navigate once with page scripts blocked to seed per-tab sessionStorage on exact origin
   550→  const completePromise = waitForTabComplete(tab.id, 9000);
   551→  await chrome.tabs.update(tab.id, { url: targetUrl });
   552→  await completePromise;
   553→
   554→  // Prepare non-HttpOnly cookies map for document.cookie virtualization
   555→  const jsCookies = {};
   556→  for (const c of account.cookies || []) {
   557→    if (c && !c.httpOnly && c.name) {
   558→      jsCookies[c.name] = c.value ?? '';
   559→    }
   560→  }
   561→
   562→  const bootPayload = {
   563→    siteKey: record.siteKey,
   564→    accountId: account.id,
   565→    accountName: account.name,
   566→    shortBadge: toPersianDigits(accIdx + 1),
   567→    color: account.color || '#06b6d4',
   568→    localStorage: account.localStorage || {},
   569→    sessionStorage: account.sessionStorage || {},
   570→    jsCookies
   571→  };
   572→
   573→  await chrome.scripting
   574→    .executeScript({
   575→      target: { tabId: tab.id },
   576→      func: (payload) => {
   577→        try {
   578→          window.sessionStorage.setItem('__AI_SWITCHER_ISO__', JSON.stringify(payload));
   579→        } catch {}
   580→      },
   581→      args: [bootPayload]
   582→    })
   583→    .catch(() => {});
   584→
   585→  // 3. Remove temporary CSP script-blocker and reload tab for real execution!
   586→  await chrome.declarativeNetRequest
   587→    .updateSessionRules({ removeRuleIds: [bootCspRuleId] })
   588→    .catch(() => {});
   589→
   590→  const isoEntry = isolatedTabs.get(tab.id);
   591→  if (isoEntry) {
   592→    isoEntry.booting = false;
   593→    await persistIsolatedSession();
   594→  }
   595→
   596→  // Update usage stats
   597→  await withSiteLock(record.siteKey, async () => {
   598→    const freshRec = await getDomainRecord(record.siteKey);
   599→    const targetAcc = freshRec.accounts.find((a) => a.id === account.id);
   600→    if (targetAcc) {
   601→      targetAcc.lastUsedAt = Date.now();

   618→// Observe Set-Cookie headers on Isolated Tabs so rotated tokens stay fresh!
   619→const pendingCookieSaves = new Map();

   621→function parseSetCookieHeader(rawHeader, fallbackHost) {
   622→  if (!rawHeader || typeof rawHeader !== 'string') return null;
   623→  const parts = rawHeader.split(';').map((s) => s.trim());
   624→  const first = parts[0] || '';
   625→  const eq = first.indexOf('=');
   626→  if (eq <= 0) return null;
   627→
   628→  const name = first.slice(0, eq).trim();
   629→  const value = first.slice(eq + 1);
   630→  let domain = fallbackHost;
   631→  let hostOnly = true;
   632→  let path = '/';
   633→  let secure = true;
   634→  let httpOnly = false;
   635→  let expired = false;
   636→  let expirationDate;
   637→
   638→  for (let i = 1; i < parts.length; i++) {
   639→    const seg = parts[i];
   640→    const low = seg.toLowerCase();
   641→    if (low === 'httponly') {
   642→      httpOnly = true;
   643→    } else if (low === 'secure') {
   644→      secure = true;
   645→    } else if (low.startsWith('domain=')) {
   646→      const d = seg.slice(7).trim().toLowerCase();
   647→      if (d) {
   648→        domain = d.startsWith('.') ? d : '.' + d;
   649→        hostOnly = false;
   650→      }
   651→    } else if (low.startsWith('path=')) {
   652→      path = seg.slice(5).trim() || '/';
   653→    } else if (low.startsWith('max-age=')) {
   654→      const sec = Number(seg.slice(8).trim());
   655→      if (!Number.isNaN(sec)) {
   656→        if (sec <= 0) expired = true;
   657→        else expirationDate = Math.floor(Date.now() / 1000) + sec;
   658→      }
   659→    } else if (low.startsWith('expires=')) {
   660→      const ms = Date.parse(seg.slice(8).trim());
   661→      if (!Number.isNaN(ms)) {
   662→        if (ms <= Date.now()) expired = true;
   663→        else expirationDate = Math.floor(ms / 1000);
   664→      }
   665→    }
   666→  }
   667→
   668→  return {
   669→    name,
   670→    value,
   671→    domain,
   672→    hostOnly,
   673→    path,
   674→    secure,
   675→    httpOnly,
   676→    sameSite: 'lax',
   677→    session: !expirationDate,
   678→    ...(expirationDate ? { expirationDate } : {}),
   679→    expired
   680→  };
   681→}
   682→
   683→if (chrome.webRequest?.onHeadersReceived) {
   684→  chrome.webRequest.onHeadersReceived.addListener(
   685→    (details) => {
   686→      if (details.tabId < 0) return;
   687→      const iso = isolatedTabs.get(details.tabId);
   688→      if (!iso || iso.booting) return;
   689→
   690→      const setCookieHeaders = (details.responseHeaders || []).filter(
   691→        (h) => h.name && h.name.toLowerCase() === 'set-cookie' && h.value
   692→      );
   693→      if (setCookieHeaders.length === 0) return;
   694→
   695→      let reqHost = iso.siteKey;
   696→      try {
   697→        reqHost = new URL(details.url).hostname.toLowerCase();
   698→      } catch {}
   699→
   700→      const parsedList = setCookieHeaders
   701→        .map((h) => parseSetCookieHeader(h.value, reqHost))
   702→        .filter(Boolean);
   703→      if (parsedList.length === 0) return;
   704→
   705→      // Queue async update for this isolated account
   706→      const key = `${details.tabId}:${iso.accountId}`;
   707→      const existing = pendingCookieSaves.get(key) || [];
   708→      existing.push(...parsedList);
   709→      pendingCookieSaves.set(key, existing);
   710→
   711→      setTimeout(() => {
   712→        const batch = pendingCookieSaves.get(key);
   713→        if (!batch || batch.length === 0) return;
   714→        pendingCookieSaves.delete(key);
   715→
   716→        withSiteLock(iso.siteKey, async () => {
   717→          const rec = await getDomainRecord(iso.siteKey);
   718→          const acc = rec.accounts.find((a) => a.id === iso.accountId);
   719→          if (!acc) return;
   720→
   721→          const cookies = Array.isArray(acc.cookies) ? [...acc.cookies] : [];
   722→          for (const item of batch) {
   723→            const idx = cookies.findIndex(
   724→              (c) =>
   725→                c.name === item.name &&
   726→                c.domain.replace(/^\./, '') === item.domain.replace(/^\./, '')
   727→            );
   728→            if (item.expired) {
   729→              if (idx >= 0) cookies.splice(idx, 1);
   730→            } else {
   731→              const cleanObj = { ...item };
   732→              delete cleanObj.expired;
   733→              if (idx >= 0) cookies[idx] = { ...cookies[idx], ...cleanObj };
   734→              else cookies.push(cleanObj);
   735→            }
   736→          }
   737→
   738→          acc.cookies = cookies;
   739→          acc.updatedAt = Date.now();
   740→          await saveDomainRecord(iso.siteKey, rec);
   741→          await refreshIsolatedTabDnrRules(details.tabId, acc);
   742→        });
   743→      }, 250);
   744→    },
   745→    { urls: ['<all_urls>'] },
   746→    ['responseHeaders', 'extraHeaders']
   747→  );
   748→}

   754→async function captureTabStorage(tabId) {
   755→  if (!tabId) {
   756→    return { localStorage: {}, sessionStorage: {}, indexedDB: {} };
   757→  }
   758→  try {
   759→    const results = await chrome.scripting.executeScript({
   760→      target: { tabId },
   761→      func: async () => {
   762→        const MAX_ITEM_LEN = 512 * 1024;
   763→        const ls = {};
   764→        const ss = {};
   765→
   766→        try {
   767→          for (let i = 0; i < window.localStorage.length; i++) {
   768→            const k = window.localStorage.key(i);
   769→            if (k !== null && k !== '__AI_SWITCHER_ISO__') {
   770→              const v = window.localStorage.getItem(k);
   771→              if (typeof v === 'string' && v.length <= MAX_ITEM_LEN) {
   772→                ls[k] = v;
   773→              }
   774→            }
   775→          }
   776→        } catch {}
   777→
   778→        try {
   779→          for (let i = 0; i < window.sessionStorage.length; i++) {
   780→            const k = window.sessionStorage.key(i);
   781→            if (k !== null && k !== '__AI_SWITCHER_ISO__') {
   782→              const v = window.sessionStorage.getItem(k);
   783→              if (typeof v === 'string' && v.length <= MAX_ITEM_LEN) {
   784→                ss[k] = v;
   785→              }
   786→            }
   787→          }
   788→        } catch {}
   789→
   790→        const idbDump = {};
   791→        try {
   792→          if (window.indexedDB && typeof window.indexedDB.databases === 'function') {
   793→            const captureIdb = async () => {
   794→              const dbs = await window.indexedDB.databases();
   795→              const authPattern = /firebase|auth|session|token|supabase|clerk|keyval|user/i;
   796→              for (const info of dbs) {
   797→                if (!info.name || !authPattern.test(info.name)) continue;
   798→                await new Promise((resolve) => {
   799→                  let settled = false;
   800→                  const done = () => {
   801→                    if (!settled) {
   802→                      settled = true;
   803→                      resolve();
   804→                    }
   805→                  };
   806→                  const timer = setTimeout(done, 400);
   807→                  try {
   808→                    const req = window.indexedDB.open(info.name);
   809→                    req.onerror = () => {
   810→                      clearTimeout(timer);
   811→                      done();
   812→                    };
   813→                    req.onsuccess = () => {
   814→                      const db = req.result;
   815→                      const storeNames = Array.from(db.objectStoreNames || []);
   816→                      if (storeNames.length === 0) {
   817→                        db.close();
   818→                        clearTimeout(timer);
   819→                        done();
   820→                        return;
   821→                      }
   822→                      const dbRecord = {};
   823→                      try {
   824→                        const tx = db.transaction(storeNames, 'readonly');
   825→                        let pending = storeNames.length;
   826→                        for (const sName of storeNames) {
   827→                          const store = tx.objectStore(sName);
   828→                          const usesInlineKey = store.keyPath !== null;
   829→                          const valReq = store.getAll();
   830→                          const keyReq = store.getAllKeys();
   831→                          let vals = null;
   832→                          let keys = null;
   833→                          const checkStoreDone = () => {
   834→                            if (vals !== null && keys !== null) {
   835→                              try {
   836→                                const jsonSize = JSON.stringify(vals).length;
   837→                                if (jsonSize <= 256 * 1024) {
   838→                                  dbRecord[sName] = {
   839→                                    usesInlineKey,
   840→                                    entries: vals.map((v, idx) => ({
   841→                                      key: keys[idx],
   842→                                      value: v
   843→                                    }))
   844→                                  };
   845→                                }
   846→                              } catch {}
   847→                              pending--;
   848→                              if (pending <= 0) {
   849→                                idbDump[info.name] = dbRecord;
   850→                                db.close();
   851→                                clearTimeout(timer);
   852→                                done();
   853→                              }
   854→                            }
   855→                          };
   856→                          valReq.onsuccess = () => {
   857→                            vals = valReq.result || [];
   858→                            checkStoreDone();
   859→                          };
   860→                          keyReq.onsuccess = () => {
   861→                            keys = keyReq.result || [];
   862→                            checkStoreDone();
   863→                          };
   864→                          valReq.onerror = keyReq.onerror = () => {
   865→                            vals = [];
   866→                            keys = [];
   867→                            checkStoreDone();
   868→                          };
   869→                        }
   870→                      } catch {
   871→                        db.close();
   872→                        clearTimeout(timer);
   873→                        done();
   874→                      }
   875→                    };
   876→                  } catch {
   877→                    clearTimeout(timer);
   878→                    done();
   879→                  }
   880→                });
   881→              }
   882→            };
   883→            await Promise.race([
   884→              captureIdb(),
   885→              new Promise((r) => setTimeout(r, 700))
   886→            ]);
   887→          }
   888→        } catch {}
   889→
   890→        return { localStorage: ls, sessionStorage: ss, indexedDB: idbDump };
   891→      }
   892→    });
   893→
   894→    if (results && results[0] && results[0].result) {
   895→      return results[0].result;
   896→    }
   897→  } catch {}
   898→  return { localStorage: {}, sessionStorage: {}, indexedDB: {} };
   899→}

   901→async function applyTabStorageAndReload(tabId, storagePayload, includeStorage) {
   902→  if (!tabId) return;
   903→
   904→  if (includeStorage) {
   905→    try {
   906→      await chrome.scripting.executeScript({
   907→        target: { tabId },
   908→        func: async (payload) => {
   909→          try {
   910→            window.localStorage.clear();
   911→            const ls = (payload && payload.localStorage) || {};
   912→            for (const [k, v] of Object.entries(ls)) {
   913→              window.localStorage.setItem(k, v);
   914→            }
   915→          } catch {}
   916→
   917→          try {
   918→            window.sessionStorage.clear();
   919→            const ss = (payload && payload.sessionStorage) || {};
   920→            for (const [k, v] of Object.entries(ss)) {
   921→              window.sessionStorage.setItem(k, v);
   922→            }
   923→          } catch {}
   924→
   925→          try {
   926→            if (window.indexedDB && typeof window.indexedDB.databases === 'function') {
   927→              const syncIdb = async () => {
   928→                const targetIdb = (payload && payload.indexedDB) || {};
   929→                const existingDbs = await window.indexedDB.databases();
   930→                const authPattern = /firebase|auth|session|token|supabase|clerk|keyval|user/i;
   931→                const dbNames = new Set([
   932→                  ...Object.keys(targetIdb),
   933→                  ...existingDbs
   934→                    .map((d) => d.name)
   935→                    .filter((n) => n && authPattern.test(n))
   936→                ]);
   937→
   938→                for (const dbName of dbNames) {
   939→                  await new Promise((resolve) => {
   940→                    let settled = false;
   941→                    const done = () => {
   942→                      if (!settled) {
   943→                        settled = true;
   944→                        resolve();
   945→                      }
   946→                    };
   947→                    const timer = setTimeout(done, 400);
   948→                    try {
   949→                      const req = window.indexedDB.open(dbName);
   950→                      req.onerror = () => {
   951→                        clearTimeout(timer);
   952→                        done();
   953→                      };
   954→                      req.onsuccess = () => {
   955→                        const db = req.result;
   956→                        const storeNames = Array.from(db.objectStoreNames || []);
   957→                        if (storeNames.length === 0) {
   958→                          db.close();
   959→                          clearTimeout(timer);
   960→                          done();
   961→                          return;
   962→                        }
   963→                        try {
   964→                          const tx = db.transaction(storeNames, 'readwrite');
   965→                          const savedStores = targetIdb[dbName] || {};
   966→                          for (const sName of storeNames) {
   967→                            const store = tx.objectStore(sName);
   968→                            store.clear();
   969→                            const savedStore = savedStores[sName];
   970→                            if (savedStore && Array.isArray(savedStore.entries)) {
   971→                              for (const item of savedStore.entries) {
   972→                                try {
   973→                                  if (savedStore.usesInlineKey || store.keyPath !== null) {
   974→                                    store.put(item.value);
   975→                                  } else {
   976→                                    store.put(item.value, item.key);
   977→                                  }
   978→                                } catch {}
   979→                              }
   980→                            }
   981→                          }
   982→                          tx.oncomplete = tx.onerror = tx.onabort = () => {
   983→                            db.close();
   984→                            clearTimeout(timer);
   985→                            done();
   986→                          };
   987→                        } catch {
   988→                          db.close();
   989→                          clearTimeout(timer);
   990→                          done();
   991→                        }
   992→                      };
   993→                    } catch {
   994→                      clearTimeout(timer);
   995→                      done();
   996→                    }
   997→                  });
   998→                }
   999→              };
  1000→              await Promise.race([
  1001→                syncIdb(),
  1002→                new Promise((r) => setTimeout(r, 750))
  1003→              ]);
  1004→            }
  1005→          } catch {}
  1006→        },
  1007→        args: [storagePayload || { localStorage: {}, sessionStorage: {}, indexedDB: {} }]
  1008→      });
  1009→
  1010→      await chrome.scripting
  1011→        .executeScript({
  1012→          target: { tabId },
  1013→          world: 'MAIN',
  1014→          func: () => {
  1015→            try {
  1016→              Storage.prototype.setItem = function () {};
  1017→              Storage.prototype.removeItem = function () {};
  1018→              Storage.prototype.clear = function () {};
  1019→            } catch {}
  1020→          }
  1021→        })
  1022→        .catch(() => {});
  1023→    } catch {}
  1024→  }
  1025→
  1026→  await chrome.tabs.reload(tabId);
  1027→}

  1033→async function updateBadgeForTab(tabId, urlStr) {

  1101→async function maybeAutoSyncOutgoingAccount(record, siteInfo, tabId, storeId) {
  1102→  if (!record.settings.autoSync || !record.activeAccountId) return;
  1103→  // Do not overwrite shared activeAccountId from an isolated tab
  1104→  if (isolatedTabs.has(tabId)) return;
  1105→
  1106→  const outgoing = record.accounts.find((a) => a.id === record.activeAccountId);
  1107→  if (!outgoing) return;
  1108→
  1109→  const liveCookies = await captureCookies(siteInfo.cookieDomains, storeId);
  1110→  const prevCount = Array.isArray(outgoing.cookies) ? outgoing.cookies.length : 0;
  1111→
  1112→  if (liveCookies.length > 0 && liveCookies.length >= Math.floor(prevCount * 0.5)) {
  1113→    outgoing.cookies = liveCookies;
  1114→    if (record.settings.includeStorage) {
  1115→      const liveStorage = await captureTabStorage(tabId);
  1116→      const prevLsCount = Object.keys(outgoing.localStorage || {}).length;
  1117→      const liveLsCount = Object.keys(liveStorage.localStorage || {}).length;
  1118→      if (prevLsCount === 0 || liveLsCount >= Math.floor(prevLsCount * 0.5)) {
  1119→        outgoing.localStorage = liveStorage.localStorage;
  1120→        outgoing.sessionStorage = liveStorage.sessionStorage;
  1121→        outgoing.indexedDB = liveStorage.indexedDB;
  1122→      }
  1123→    }
  1124→    outgoing.updatedAt = Date.now();
  1125→  }
  1126→}

  1377→function extractOtpCode(text) {

  1382→  );

  1386→  if (sixDigit) return sixDigit[1];

  1390→    const val = fiveOrFour[1];
  1391→    // Avoid matching common years like 2024-2029

  1397→function extractVerificationLinks(text, htmlArr) {
  1398→  const links = new Set();
  1399→  const combined = [text || '', ...(Array.isArray(htmlArr) ? htmlArr : [])].join('\n');
  1400→  const regex = /https?:\/\/[^\s"'<>\\)]+/gi;
  1401→  let m;
  1402→  while ((m = regex.exec(combined)) !== null) {
  1403→    const url = m[0].replace(/[.,;!?]+$/, '');
  1404→    if (
  1405→      !/\.(png|jpg|jpeg|gif|svg|css|woff2?|ico)(\?|$)/i.test(url) &&
  1406→      !url.includes('w3.org') &&
  1407→      !url.includes('schema.org')
  1408→    ) {
  1409→      links.add(url);
  1410→    }
  1411→  }
  1412→  const all = Array.from(links);
  1413→  // Sort verification/confirm/magic links to top
  1414→  all.sort((a, b) => {
  1415→    const score = (u) =>
  1416→      /verify|confirm|activate|auth|token|magic|callback|login/i.test(u) ? 0 : 1;
  1417→    return score(a) - score(b);
  1418→  });
  1419→  return all.slice(0, 6);
