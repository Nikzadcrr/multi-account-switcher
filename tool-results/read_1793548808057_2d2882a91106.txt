    83→async function copyText(text) {
    84→  try {
    85→    if (navigator.clipboard?.writeText) {
    86→      await navigator.clipboard.writeText(text);
    87→      return true;
    88→    }
    89→  } catch {}
    90→  try {
    91→    const ta = document.createElement('textarea');
    92→    ta.value = text;
    93→    ta.setAttribute('readonly', '');
    94→    ta.style.cssText = 'position:fixed;top:-999px;left:-999px;opacity:0;pointer-events:none;';
    95→    document.body.appendChild(ta);
    96→    ta.select();
    97→    ta.setSelectionRange(0, text.length);
    98→    const ok = document.execCommand('copy');
    99→    ta.remove();
   100→    if (ok) return true;
   101→  } catch {}
   102→  return false;
   103→}

   110→function formatRelativeTime(ts) {
   111→  if (!ts) return 'استفاده‌نشده';
   112→  const diffSec = Math.max(0, Math.floor((Date.now() - ts) / 1000));
   113→  if (diffSec < 45) return 'هم‌اکنون';
   114→  const mins = Math.floor(diffSec / 60);
   115→  if (mins < 60) return `${toFa(mins)} دقیقه پیش`;
   116→  const hours = Math.floor(mins / 60);
   117→  if (hours < 24) return `${toFa(hours)} ساعت پیش`;
   118→  const days = Math.floor(hours / 24);
   119→  return `${toFa(days)} روز پیش`;
   120→}

   128→      }

   130→    });

   445→function sortedAccountsForDisplay() {
   446→  const sort = listPref.sort || 'recent';
   447→  const arr = [...state.accounts];
   448→  arr.sort((a, b) => {
   449→    if (Boolean(b.pinned) !== Boolean(a.pinned)) return b.pinned ? 1 : -1;
   450→    if (sort === 'name') return String(a.name).localeCompare(String(b.name), 'fa');
   451→    if (sort === 'created') return (b.createdAt || 0) - (a.createdAt || 0);
   452→    return (b.lastUsedAt || 0) - (a.lastUsedAt || 0);
   453→  });
   454→  return arr;
   455→}

  1117→function htmlToCleanText(htmlArr, fallbackText) {
  1118→  if (fallbackText && fallbackText.trim()) return fallbackText.trim();
  1119→  if (!Array.isArray(htmlArr) || !htmlArr.length) return '(پیام بدون متن)';
  1120→  try {
  1121→    const doc = new DOMParser().parseFromString(htmlArr.join('\n'), 'text/html');
  1122→    doc.querySelectorAll('script, style, noscript, iframe, object').forEach((el) => el.remove());
  1123→    return (doc.body?.innerText || doc.body?.textContent || '').trim() || '(پیام بدون متن)';
  1124→  } catch {
  1125→    return '(خطا در نمایش متن پیام)';
  1126→  }
  1127→}
