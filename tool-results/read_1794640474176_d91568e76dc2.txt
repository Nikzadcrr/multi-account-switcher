   129→function chatExtractJson(text) {
   130→  if (!text) return null;
   131→  let s = String(text).trim();
   132→  const fence = s.match(/```(?:json)?\s*([\s\S]*?)```/i);
   133→  if (fence) s = fence[1].trim();
   134→  const start = s.indexOf('{');
   135→  const end = s.lastIndexOf('}');
   136→  if (start >= 0 && end > start) s = s.slice(start, end + 1);
   137→  try {
   138→    return JSON.parse(s);
   139→  } catch {
   140→    return null;
   141→  }
   142→}

   144→async function chatBrainCall(userText, contextLines) {
   145→  const st = await chatGetState();
   146→  const s = st.settings;
   147→  const sys = CHAT_SYSTEM_PROMPT + '\n\nوضعیت فعلی:\n' + contextLines.join('\n');
   148→  const history = st.msgs.slice(-10).map((m) => ({
   149→    role: m.role === 'user' ? 'user' : 'assistant',
   150→    content: String(m.text || '').slice(0, 600)
   151→  }));
   152→
   153→  // فاصلهٔ حداقلی برای پولینیشنز
   154→  const waitMs = chatLastBrainAt + CHAT_MIN_BRAIN_GAP_MS - Date.now();
   155→  if (s.brainProvider === 'pollinations' && waitMs > 0) {
   156→    await agentSleep(Math.min(waitMs, 20000));
   157→  }
   158→  chatLastBrainAt = Date.now();
   159→
   160→  const controller = new AbortController();
   161→  const timer = setTimeout(() => controller.abort(), CHAT_BRAIN_TIMEOUT_MS);
   162→  try {
   163→    let text = '';
   164→    if (s.brainProvider === 'groq' && s.groqKey) {
   165→      const res = await fetch('https://api.groq.com/openai/v1/chat/completions', {
   166→        method: 'POST',
   167→        signal: controller.signal,
   168→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${s.groqKey}` },
   169→        body: JSON.stringify({
   170→          model: s.groqModel || 'llama-3.3-70b-versatile',
   171→          temperature: 0.2,
   172→          max_tokens: 400,
   173→          messages: [{ role: 'system', content: sys }, ...history]
   174→        })
   175→      });
   176→      const j = await res.json();
   177→      text = j?.choices?.[0]?.message?.content || '';
   178→    } else if (s.brainProvider === 'gemini' && s.geminiKey) {
   179→      const res = await fetch(
   180→        `https://generativelanguage.googleapis.com/v1beta/models/${s.geminiModel || 'gemini-2.0-flash'}:generateContent?key=${s.geminiKey}`,
   181→        {
   182→          method: 'POST',
   183→          signal: controller.signal,
   184→          headers: { 'Content-Type': 'application/json' },
   185→          body: JSON.stringify({
   186→            systemInstruction: { parts: [{ text: sys }] },
   187→            contents: history.map((m) => ({ role: m.role === 'assistant' ? 'model' : 'user', parts: [{ text: m.content }] })),
   188→            generationConfig: { temperature: 0.2, maxOutputTokens: 400 }
   189→          })
   190→        }
   191→      );
   192→      const j = await res.json();
   193→      text = j?.candidates?.[0]?.content?.parts?.map((p) => p.text).join('') || '';
   194→    } else if (s.brainProvider === 'openrouter' && s.openrouterKey) {
   195→      const res = await fetch('https://openrouter.ai/api/v1/chat/completions', {
   196→        method: 'POST',
   197→        signal: controller.signal,
   198→        headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${s.openrouterKey}` },
   199→        body: JSON.stringify({
   200→          model: s.openrouterModel || 'meta-llama/llama-3.3-70b-instruct:free',
   201→          max_tokens: 400,
   202→          messages: [{ role: 'system', content: sys }, ...history]
   203→        })
   204→      });
   205→      const j = await res.json();
   206→      text = j?.choices?.[0]?.message?.content || '';
   207→    } else {
   208→      // پیش‌فرض: Pollinations — رایگان و بدون کلید
   209→      const res = await fetch('https://text.pollinations.ai/', {
   210→        method: 'POST',
   211→        signal: controller.signal,
   212→        headers: { 'Content-Type': 'application/json' },
   213→        body: JSON.stringify({
   214→          model: s.brainModel || 'openai',
   215→          messages: [{ role: 'system', content: sys }, ...history]
   216→        })
   217→      });
   218→      text = await res.text();
   219→    }
   220→    clearTimeout(timer);
   221→    return { raw: text, json: chatExtractJson(text) };
   222→  } catch (e) {
   223→    clearTimeout(timer);
   224→    return { raw: '', json: null, error: e?.message || 'brain_error' };
   225→  }
   226→}

   228→function chatLocalParse(userText) {
   229→  const t = String(userText || '').toLowerCase().trim();
   230→  const out = { intent: 'talk', site: '', count: 1, freshMail: true, reply: '', _local: true };
   231→
   232→  if (/(لغو|نگه|متوقف|stop|cancel|بس کن)/.test(t)) {
   233→    out.intent = 'cancel';
   234→    return out;
   235→  }
   236→  if (/(وضعیت|چقدر مونده|status|progress|کجا رسید)/.test(t)) {
   237→    out.intent = 'status';
   238→    return out;
   239→  }
   240→  if (/(کمک|راهنما|چطوری|help|چطور کار)/.test(t)) {
   241→    out.intent = 'help';
   242→    return out;
   243→  }
   244→
   245→  // تشخیص قصد ثبت‌نام: «بساز/ثبت‌نام/اکانت/reg»
   246→  const wantsRegister = /(بساز|ثبت[‌‌ ]?نام|اکانت[‌‌ ]?(بساز|بگیر)|ساخت اکانت|reg(ister)?|account|signup|sign up)/.test(t);
   247→
   248→  // سایت: KB + دامنهٔ صریح
   249→  let site = '';
   250→  for (const item of CHAT_SITE_KB) {
   251→    if (item.keys.some((k) => t.includes(k))) {
   252→      site = item.domain;
   253→      break;
   254→    }
   255→  }
   256→  if (!site) {
   257→    const url = t.match(/([a-z0-9-]+\.(?:com|ai|io|net|org|app|dev|co|me)[a-z0-9./-]*)/i);
   258→    if (url) site = url[1].replace(/\/+$/, '');
   259→  }
   260→
   261→  // تعداد
   262→  let count = 1;
   263→  const numMap = { یک: 1, دو: 2, دوتا: 2, سه: 3, سه‌تا: 3, چهار: 4, پنج: 5, شش: 6, هفت: 7, هشت: 8, نه: 9, ده: 10 };
   264→  const m2 = t.match(/(\d+)\s*(تا|عدد|اکانت|account)?/);
   265→  const m3 = t.match(/([\u06F0-\u06F9]{1,2})/);
   266→  if (m2 && parseInt(m2[1], 10) > 0) count = Math.min(20, parseInt(m2[1], 10));
   267→  else if (m3) {
   268→    const fa = m3[1].replace(/[\u06F0-\u06F9]/g, (d) => String('۰۱۲۳۴۵۶۷۸۹'.indexOf(d)));
   269→    const v = parseInt(fa, 10);
   270→    if (v > 0) count = Math.min(20, v);
   271→  } else {
   272→    for (const [w, v] of Object.entries(numMap)) {
   273→      if (new RegExp(`${w}(\\s*(تا|عدد|اکانت|account))`).test(t)) {
   274→        count = v;
   275→        break;
   276→      }
   277→    }
   278→  }
   279→
   280→  if (wantsRegister || site) out.intent = 'register';
   281→  out.site = site;
   282→  out.count = count;
   283→  return out;
   284→}

   315→function chatNormalizeSite(site) {
   316→  let s = String(site || '').trim().toLowerCase();
   317→  if (!s) return null;
   318→  s = s.replace(/^https?:\/\//, '').replace(/^www\./, '').split(/[/?#]/)[0];
   319→  if (!/^[a-z0-9.-]+\.[a-z]{2,}$/.test(s)) return null;
   320→  return s;
   321→}

   368→async function chatEnsureSignupPage(tabId, run, domain, startUrl) {
   369→  // اگر تب روی دامنه است، از همان صفحه شروع کن؛ وگرنه مقصد را باز کن
   370→  let tab = await chrome.tabs.get(tabId).catch(() => null);
   371→  let onDomain = false;
   372→  try {
   373→    onDomain = tab && new URL(tab.url).hostname.replace(/^www\./, '') === domain;
   374→  } catch {}
   375→
   376→  const candidates = [];
   377→  if (startUrl) candidates.push(startUrl);
   378→  if (onDomain && tab.url && !candidates.includes(tab.url)) candidates.unshift(tab.url);
   379→  if (domain && !candidates.some((u) => u.startsWith('https://'))) {
   380→    candidates.push(`https://${domain}/`);
   381→  }
   382→  for (const h of CHAT_SIGNUP_HINTS) candidates.push(`https://${domain}${h}`);
   383→
   384→  for (let i = 0; i < Math.min(candidates.length, 7); i++) {
   385→    if (run.cancelled) return null;
   386→    const url = candidates[i];
   387→    if (!onDomain || i > 0) {
   388→      chatPush('system', `رفتن به ${url}`);
   389→      try {
   390→        await chrome.tabs.update(tabId, { url });
   391→      } catch {
   392→        return null;
   393→      }
   394→      tab = await chatWaitTabComplete(tabId, 22000);
   395→      if (!tab) return null;
   396→      onDomain = true;
   397→    }
   398→    await agentSleep(700);
   399→    const scan = await chatScanForForm(tabId);
   400→    if (scan?.ok && scan.hasFields) {
   401→      return { ok: true, scan, url: tab.url };
   402→    }
   403→  }
   404→  return null;
   405→}

   432→    if (solved) {
   433→      chatPush('system', 'اسلایدر حل شد ✓ — دوباره فرم ارسال می‌شود.');
   434→      await agentSleep(1200);
   435→      // اگر صفحهٔ چالش مستقل بود، بعد از حل برمی‌گردد؛ فرم را دوباره پر و بفرست
   436→      try {
   437→        await agentSendTab(tabId, { type: 'AGENT_CLICK_SUBMIT' });
   438→      } catch {}
   439→      return true;
   440→    }
   441→    chatPush('system', 'حل خودکار اسلایدر کامل نشد؛ یک بار دستی حل کن و «ادامه» را نزن — خودم چک می‌کنم.');
   442→    return false;
   443→  } catch {
   444→    return false;
   445→  }
   446→}

   452→function chatWaitRun(tabId, timeoutMs = 420000) {
   453→  return new Promise((resolve) => {
   454→    const t0 = Date.now();
   455→    const timer = setInterval(async () => {
   456→      const run = typeof agentRuns !== 'undefined' ? agentRuns.get(tabId) : null;
   457→      if (!run) {
   458→        clearInterval(timer);
   459→        return resolve({ status: 'failed', error: 'run_missing' });
   460→      }
   461→      if (['done', 'failed', 'cancelled'].includes(run.status)) {
   462→        clearInterval(timer);
   463→        return resolve({ status: run.status, run });
   464→      }
   465→      if (Date.now() - t0 > timeoutMs) {
   466→        clearInterval(timer);
   467→        run.status = 'cancelled';
   468→        return resolve({ status: 'timeout', run });
   469→      }
   470→    }, 900);
   471→  });
   472→}

   474→async function chatRegisterOneAccount(job, index) {
   475→  const domain = chatResolveSite(job.site) || chatNormalizeSite(job.site);
   476→  if (!domain) {
   477→    await chatPush('agent', 'دامنهٔ سایت معتبر نیست. مثل «chat.qwen.ai» بنویس.');
   478→    return { ok: false, error: 'bad_site' };
   479→  }
   480→
   481→  const kb = chatFindKB(job.site) || null;
   482→  await chatPush('system', `— اکانت ${index}/${job.count}: ${kb?.label || domain}`);
   483→
   484→  // ۱) تب مقصد
   485→  let tab = await chatFindSiteTab(domain);
   486→  if (!tab) {
   487→    tab = await chrome.tabs
   488→      .create({ url: kb?.start || `https://${domain}/`, active: false })
   489→      .then((t) => chrome.tabs.get(t.id))
   490→      .catch(() => null);
   491→  }
   492→  if (!tab?.id) {
   493→    await chatPush('agent', 'تب سایت باز نشد.');
   494→    return { ok: false, error: 'no_tab' };
   495→  }
   496→  await chatWaitTabComplete(tab.id, 20000);
   497→  await chrome.tabs.update(tab.id, { active: true }).catch(() => {});
   498→
   499→  // ۲) پیدا کردن فرم ثبت‌نام
   500→  const found = await chatEnsureSignupPage(tab.id, job, domain, kb?.start);
   501→  if (!found) {
   502→    await chatPush(
   503→      'agent',
   504→      `فرم ثبت‌نام در ${kb?.label || domain} پیدا نشد. یک بار دستی صفحهٔ ثبت‌نام را باز کن و بگو «ادامه بده» — یا آدرس دقیق صفحه را بفرست.`
   505→    );
   506→    job.awaitingUrlHint = true;
   507→    await chatSetJob(job);
   508→    return { ok: false, error: 'no_form', needsHint: true };
   509→  }
   510→
   511→  // ۳) ایمیل موقت تازه برای هر اکانت
   512→  let mailbox = null;
   513→  const mailState = await getMailState();
   514→  if (job.freshMail === false && mailState.activeMailboxId) {
   515→    mailbox = mailState.mailboxes.find((m) => m.id === mailState.activeMailboxId) || null;
   516→  }
   517→  if (!mailbox) {
   518→    try {
   519→      mailbox = await createTempMailbox(job.mailProvider || 'hd');
   520→      await chatPush('system', `ایمیل موقت ساخته شد: ${mailbox.email}`);
   521→    } catch (e) {
   522→      await chatPush('agent', `ساخت ایمیل موقت شکست خورد: ${e?.message || '?'}`);
   523→      return { ok: false, error: 'mail_fail' };
   524→    }
   525→  }
   526→
   527→  // baseline برای تشخیص ایمیل تازه
   528→  let baselineIds = [];
   529→  try {
   530→    const baseMsgs = await listMessagesForMailbox(mailbox);
   531→    baselineIds = (baseMsgs || []).map((m) => String(m.id));
   532→  } catch {}
   533→
   534→  // ۴) اجرای فلو اصلی
   535→  const hostname = domain;
   536→  await runAgentFlow(tab.id, {
   537→    email: mailbox.email,
   538→    mailboxId: mailbox.id,
   539→    username: agentGenUsername(),
   540→    password: agentStrongPassword(),
   541→    autoSubmit: true,
   542→    autoSave: true,
   543→    baselineIds,
   544→    hostname,
   545→    _chatJob: job.id
   546→  });
   547→
   548→  const result = await chatWaitRun(tab.id);
   549→
   550→  // ۵) پس از submit: چالش اسلایدر را چک و حل کن و دوباره فرم بفرست
   551→  if (result.status === 'running' || result.status === 'waiting_otp' || result.status === 'done') {
   552→    await chatPostSubmitChallenge(result.run || { cancelled: false }, tab.id);
   553→  }
   554→
   555→  const finalRun = (typeof agentRuns !== 'undefined' ? agentRuns.get(tab.id) : null) || result.run || {};
   556→  const ok = finalRun.status === 'done' && (finalRun.savedAccountId || finalRun.otpFilled || finalRun.otp);
   557→  return {
   558→    ok,
   559→    status: finalRun.status,
   560→    email: finalRun.email,
   561→    password: finalRun.password,
   562→    savedAccountId: finalRun.savedAccountId || null,
   563→    otp: finalRun.otp || null,
   564→    error: finalRun.error || null
   565→  };
   566→}

   572→async function chatStartJob(plan) {
   573→  if (chatJobActive) {
   574→    await chatPush('agent', 'یک کار فعال در جریان است؛ اول با «لغو کن» تمامش کن.');
   575→    return { ok: false, error: 'busy' };
   576→  }
   577→  const domain = chatResolveSite(plan.site);
   578→  if (!domain) {
   579→    await chatPush('agent', 'سایت را دقیق‌تر بگو — مثلاً: «۲ تا اکانت qwen بساز» یا آدرس کامل بده.');
   580→    return { ok: false, error: 'no_site' };
   581→  }
   582→
   583→  const kb = chatFindKB(domain) || { domain, label: domain, start: null };
   584→  const job = {
   585→    id: `job_${Date.now()}`,
   586→    site: domain,
   587→    siteLabel: kb.label,
   588→    startUrl: kb.start,
   589→    count: Math.max(1, Math.min(20, plan.count || 1)),
   590→    done: 0,
   591→    failed: 0,
   592→    attempts: {},
   593→    freshMail: plan.freshMail !== false,
   594→    mailProvider: plan.mailProvider || (await chatGetState()).settings.mailProvider || 'hd',
   595→    cancelled: false,
   596→    startedAt: Date.now(),
   597→    results: []
   598→  };
   599→  chatJobActive = true;
   600→  await chatSetJob(job);
   601→  await chatPush(
   602→    'agent',
   603→    `باشه! شروع می‌کنم: ${job.count} اکانت روی ${kb.label} با ایمیل موقت. هر مرحله را همین‌جا می‌گویم. 🚀`
   604→  );
   605→
   606→  runChatJob(job).finally(() => {
   607→    chatJobActive = false;
   608→  });
   609→  return { ok: true };
   610→}

   612→async function runChatJob(job) {
   613→  try {
   614→    for (let i = job.done + job.failed + 1; i <= job.count && !job.cancelled; i++) {
   615→      let attempt = 0;
   616→      let result = null;
   617→      while (attempt < 3 && !job.cancelled) {
   618→        attempt++;
   619→        job.attempts[i] = attempt;
   620→        await chatSetJob(job);
   621→        result = await chatRegisterOneAccount(job, i);
   622→        if (result.ok) break;
   623→        if (result.needsHint) {
   624→          // منتظر راهنمای URL از کاربر — حلقهٔ بیرونی هم باید بایستد
   625→          job.pausedForHint = i;
   626→          await chatSetJob(job);
   627→          await chatPush('system', 'کار متوقف شد تا آدرس دقیق صفحهٔ ثبت‌نام را بدهی. بنویس: «ادامه بده <آدرس>»');
   628→          return;
   629→        }
   630→        if (attempt < 3) {
   631→          await chatPush('system', `تلاش ${attempt} ناموفق بود؛ با ایمیل تازه دوباره…`);
   632→          await agentSleep(2500);
   633→        }
   634→      }
   635→      if (job.cancelled) break;
   636→
   637→      if (result && result.ok) {
   638→        job.done++;
   639→        job.results.push({ index: i, ok: true, email: result.email, password: result.password, otp: result.otp });
   640→        await chatPush('agent', `✅ اکانت ${i} ساخته شد: ${result.email} — رمز در خزانه ذخیره شد.`);
   641→        if (result.otp) await chatPush('system', `کد تأیید: ${result.otp}`);
   642→      } else {
   643→        job.failed++;
   644→        job.results.push({ index: i, ok: false, error: result?.error || 'unknown' });
   645→        await chatPush('agent', `❌ اکانت ${i} بعد از ۳ تلاش نشد${result?.error ? ` (${result.error})` : ''}. بقیه را ادامه می‌دهم.`);
   646→      }
   647→      await chatSetJob(job);
   648→      await agentSleep(1800);
   649→    }
   650→
   651→    const totalOk = job.done;
   652→    if (job.cancelled) {
   653→      await chatPush('agent', `کار لغو شد. تا اینجا: ${totalOk} موفق از ${job.done + job.failed} اجرا.`);
   654→    } else if (job.done + job.failed >= job.count) {
   655→      const summary = job.results
   656→        .filter((r) => r.ok)
   657→        .map((r) => `• ${r.email}${r.otp ? ` (کد: ${r.otp})` : ''}`)
   658→        .join('\n');
   659→      await chatPush(
   660→        'agent',
   661→        summary
   662→          ? `تمام شد! ${totalOk}/${job.count} اکانت ساخته و در خزانه ذخیره شد:\n${summary}`
   663→          : `متأسفانه هیچ‌کدام از ${job.count} اکانت ساخته نشد. می‌خوای دوباره امتحان کنیم؟`
   664→      );
   665→    }
   666→    job.finishedAt = Date.now();
   667→    await chatSetJob(job);
   668→  } catch (e) {
   669→    await chatPush('agent', `خطای غیرمنتظره: ${e?.message || e}`);
   670→    job.finishedAt = Date.now();
   671→    await chatSetJob(job);
   672→  }
   673→}

   679→async function agentChatSend(userText, forceUrlHint) {
   680→  const text = String(userText || '').trim();
   681→  if (!text) return { ok: false, error: 'empty' };
   682→  await chatPush('user', text);
   683→
   684→  const st = await chatGetState();
   685→
   686→  // حالت انتظار آدرس دقیق صفحهٔ ثبت‌نام
   687→  if (st.job?.pausedForHint || forceUrlHint) {
   688→    const url = text.match(/https?:\/\/[^\s]+/);
   689→    if (/ادامه|برو|همین|اینو|این\b/.test(text.toLowerCase()) || url) {
   690→      const job = st.job;
   691→      const hintUrl = url ? url[0] : null;
   692→      if (hintUrl) {
   693→        job.startUrl = hintUrl;
   694→        job.site = chatNormalizeSite(new URL(hintUrl).hostname) || job.site;
   695→        chatFindKB(job.site) && (job.siteLabel = chatFindKB(job.site).label);
   696→      }
   697→      const pausedIndex = job.pausedForHint || (job.done + job.failed + 1);
   698→      job.pausedForHint = null;
   699→      await chatSetJob(job);
   700→      chatJobActive = true;
   701→      runChatJob(job, { resumeIndex: pausedIndex, startUrl: hintUrl }).finally(() => {
   702→        chatJobActive = false;
   703→      });
   704→      await chatPush('agent', 'برگشتم سر کار! ادامه می‌دهم…');
   705→      return { ok: true };
   706→    }
   707→  }
   708→
   709→  const ctx = await chatBuildContextLines();
   710→  let plan = null;
   711→  try {
   712→    const brain = await chatBrainCall(text, ctx);
   713→    if (brain.json && brain.json.intent) plan = brain.json;
   714→    else if (brain.raw && brain.raw.length > 4) {
   715→      // LLM جواب داده ولی JSON نبود → برای register دوباره محلی پارس می‌کنیم
   716→      plan = chatLocalParse(text);
   717→      if (plan.intent === 'talk') {
   718→        await chatPush('agent', String(brain.raw).slice(0, 500));
   719→        return { ok: true };
   720→      }
   721→    }
   722→  } catch {}
   723→
   724→  if (!plan) plan = chatLocalParse(text);
   725→
   726→  switch (plan.intent) {
   727→    case 'register': {
   728→      // اگر مغز سایت را نداد، از پارسر محلی کمک بگیر
   729→      if (!plan.site) {
   730→        const lp = chatLocalParse(text);
   731→        if (lp.site) {
   732→          plan.site = lp.site;
   733→          if (!plan.count || plan.count < 1) plan.count = lp.count;
   734→        }
   735→        if (!plan.site) {
   736→          await chatPush('agent', 'روی کدوم سایت اکانت بسازم؟ اسم یا آدرسش را بنویس (مثلاً qwen یا chat.qwen.ai).');
   737→          return { ok: true };
   738→        }
   739→      }
   740→      if (!plan.count || plan.count < 1) plan.count = 1;
   741→      return chatStartJob(plan);
   742→    }
   743→    case 'status': {
   744→      const job = st.job;
   745→      if (!job || job.finishedAt) await chatPush('agent', 'در حال حاضر کاری در جریان نیست. بنویس «کمک» تا شروع کنیم.');
   746→      else
   747→        await chatPush(
   748→          'agent',
   749→          `کار «${job.siteLabel || job.site}»: ${job.done} موفق، ${job.failed} ناموفق از ${job.count} — ${job.pausedForHint ? 'منتظر آدرس صفحهٔ ثبت‌نام از تو هستم' : 'در جریان است'}.`
   750→        );
   751→      return { ok: true };
   752→    }
   753→    case 'cancel': {
   754→      const job = st.job;
   755→      job && (job.cancelled = true) && (await chatSetJob(job));
   756→      for (const [tabId, run] of typeof agentRuns !== 'undefined' ? agentRuns : []) {
   757→        if (['running', 'waiting_captcha_manual', 'waiting_otp'].includes(run.status)) {
   758→          run.status = 'cancelled';
   759→          if (run._manualResolve) run._manualResolve();
   760→        }
   761→      }
   762→      await chatPush('agent', 'لغو شد. هر وقت خواستی دوباره بگو.');
   763→      return { ok: true };
   764→    }
   765→    case 'help': {
   766→      await chatPush(
   767→        'agent',
   768→        'راهنما: کافیست بنویسی مثلاً «۲ تا اکانت qwen بساز» یا «chat.qwen.ai — ۳ اکانت». من صفحهٔ ثبت‌نام را باز می‌کنم، ایمیل موقت می‌سازم، فرم را پر می‌کنم، کپچا/اسلایدر را حل می‌کنم و کد تأیید را از ایمیل می‌گیرم. حساب‌ها در «خزانه» ذخیره می‌شوند. وسط کار می‌توانی «وضعیت» یا «لغو کن» بگویی.'
   769→      );
   770→      return { ok: true };
   771→    }
   772→    default: {
   773→      if (plan.reply) await chatPush('agent', plan.reply);
   774→      else
   775→        await chatPush(
   776→          'agent',
   777→          'شنیدم! اگر می‌خواهی اکانت بسازم بنویس مثل: «۳ اکانت qwen بساز» — یا «کمک» برای راهنما.'
   778→        );
   779→      return { ok: true };
   780→    }
   781→  }
   782→}

   788→chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => {
   789→  if (!msg || typeof msg.type !== 'string' || !msg.type.startsWith('AGENT_CHAT_')) return false;
   790→  (async () => {
   791→    try {

   793→        case 'AGENT_CHAT_SEND':
   794→          return sendResponse(await agentChatSend(msg.text, msg.forceUrlHint));
   795→        case 'AGENT_CHAT_STATE': {
   796→          const st = await chatGetState();
   797→          return sendResponse({ ok: true, ...st, active: chatJobActive });
   798→        }
   799→        case 'AGENT_CHAT_SAVE_SETTINGS':
   800→          return sendResponse({ ok: true, settings: await chatSaveSettings(msg.settings || {}) });
   801→        case 'AGENT_CHAT_CANCEL': {
   802→          const st = await chatGetState();
   803→          if (st.job) {
   804→            st.job.cancelled = true;
   805→            await chatSetJob(st.job);
   806→          }
   807→          for (const [tabId, run] of typeof agentRuns !== 'undefined' ? agentRuns : []) {
   808→            if (['running', 'waiting_captcha_manual', 'waiting_otp'].includes(run.status)) {
   809→              run.status = 'cancelled';
   810→              if (run._manualResolve) run._manualResolve();
   811→            }
   812→          }
   813→          chatJobActive = false;
   814→          await chatPush('agent', 'لغو شد ✓');
   815→          return sendResponse({ ok: true });
   816→        }
   817→        case 'AGENT_CHAT_CLEAR': {
   818→          await chatSaveMsgs([]);
   819→          await chatSetJob(null);
   820→          chatBroadcastReset();
   821→          await chatPush('agent', 'چت تمیز شد. بگو چی بسازم! 😊');
   822→          return sendResponse({ ok: true });
   823→        }
   824→        default:
   825→          return sendResponse({ ok: false, error: 'unknown_chat_msg' });
   826→      }
   827→    } catch (e) {
   828→      try {
   829→        sendResponse({ ok: false, error: e?.message || 'chat_error' });
   830→      } catch {}
   831→    }
   832→  })();
   833→  return true; // async sendResponse
   834→});
