    12→function toFa(val) {

    16→function formatRelativeTime(ts) {
    17→  if (!ts) return 'استفاده‌نشده';
    18→  const diffSec = Math.max(0, Math.floor((Date.now() - ts) / 1000));
    19→  if (diffSec < 45) return 'هم‌اکنون';
    20→  const mins = Math.floor(diffSec / 60);
    21→  if (mins < 60) return `${toFa(mins)} دقیقه پیش`;
    22→  const hours = Math.floor(mins / 60);
    23→  if (hours < 24) return `${toFa(hours)} ساعت پیش`;
    24→  const days = Math.floor(hours / 24);
    25→  return `${toFa(days)} روز پیش`;
    26→}

    28→function sendMsg(payload) {
    29→  return new Promise((resolve) => {
    30→    chrome.runtime.sendMessage(payload, (res) => {
    31→      if (chrome.runtime.lastError) {
    32→        resolve({ ok: false, error: chrome.runtime.lastError.message });
    33→        return;
    34→      }
    35→      resolve(res || { ok: false, error: 'پاسخی دریافت نشد.' });
    36→    });
    37→  });
    38→}

   191→async function loadState() {
   192→  const [tab] = await chrome.tabs.query({ active: true, currentWindow: true });
   193→
   194→  const [res, mRes] = await Promise.all([
   195→    sendMsg({ type: 'GET_SITE_STATE', tabId: tab?.id, url: tab?.url }),
   196→    sendMsg({ type: 'MAIL_GET_STATE' })
   197→  ]);
   198→
   199→  if (mRes?.ok) {
   200→    mailState.activeMailboxId = mRes.activeMailboxId;
   201→    mailState.mailboxes = mRes.mailboxes || [];
   202→    renderMailUI();
   203→  }
   204→
   205→  if (!res.ok || !res.supported) {
   206→    state.supported = false;
   207→    state.tabId = tab?.id || null;
   208→    els.domainBadge.textContent = 'صفحه داخلی مرورگر';
   209→    els.countBadge.textContent = '۰ اکانت';
   210→    els.isoCountBadge.classList.add('hidden');
   211→    els.unsupportedState.classList.remove('hidden');
   212→    els.mainContent.classList.add('hidden');
   213→    return;
   214→  }
   215→
   216→  state = {
   217→    ...state,
   218→    tabId: tab?.id || res.tabId,
   219→    url: tab?.url || res.url,
   220→    supported: true,
   221→    siteKey: res.siteKey,
   222→    hostname: res.hostname,
   223→    isGoogleUnified: Boolean(res.isGoogleUnified),
   224→    activeAccountId: res.activeAccountId,
   225→    isIsolatedTab: Boolean(res.isIsolatedTab),
   226→    isolatedAccountId: res.isolatedAccountId || null,
   227→    openIsolatedTabs: res.openIsolatedTabs || [],
   228→    accounts: res.accounts || [],
   229→    settings: res.settings || state.settings
   230→  };
   231→
   232→  els.unsupportedState.classList.add('hidden');
   233→  els.mainContent.classList.remove('hidden');
   234→
   235→  const currentVal = els.accountNameInput.value.trim();
   236→  if (!currentVal || /^اکانت\s+[۰-۹0-9]+$/.test(currentVal)) {
   237→    els.accountNameInput.value = `اکانت ${toFa(state.accounts.length + 1)}`;
   238→  }
   239→
   240→  renderUI();
   241→}

   243→function renderUI() {
   244→  els.domainBadge.textContent = state.siteKey;
   245→  els.countBadge.textContent = `${toFa(state.accounts.length)} اکانت`;
   246→
   247→  const openIsoCount = state.openIsolatedTabs.length;
   248→  els.isoCountBadge.classList.toggle('hidden', openIsoCount === 0);
   249→  els.isoCountBadge.textContent = `🔒 ${toFa(openIsoCount)} تب باز`;
   250→
   251→  els.googleWarning.classList.toggle('hidden', !state.isGoogleUnified);
   252→
   253→  els.chkFloatingWidget.checked = state.settings.floatingWidget !== false;
   254→  els.chkAutoSync.checked = state.settings.autoSync !== false;
   255→  els.chkIncludeStorage.checked = state.settings.includeStorage !== false;
   256→
   257→  // Link active mail indicator
   258→  const activeMb = mailState.mailboxes.find((m) => m.id === mailState.activeMailboxId);
   259→  if (activeMb) {
   260→    els.linkActiveMailRow.classList.remove('hidden');
   261→    els.linkActiveMailLabel.textContent = `اتصال ایمیل فعال (${activeMb.address}) به این اکانت`;
   262→  } else {
   263→    els.linkActiveMailRow.classList.add('hidden');
   264→  }
   265→
   266→  const total = state.accounts.length;
   267→
   268→  if (state.isIsolatedTab && state.isolatedAccountId) {
   269→    const isoIdx = state.accounts.findIndex((a) => a.id === state.isolatedAccountId);
   270→    const isoAcc = isoIdx >= 0 ? state.accounts[isoIdx] : null;
   271→    els.activeStatusText.textContent = isoAcc
   272→      ? `🔒 تب همزمان ایزوله: ${isoAcc.name} (${toFa(isoIdx + 1)})`
   273→      : '🔒 تب ایزوله';
   274→    els.activeStatusText.style.color = '#67e8f9';
   275→  } else {
   276→    const activeIdx = state.accounts.findIndex((a) => a.id === state.activeAccountId);
   277→    const activeAcc = activeIdx >= 0 ? state.accounts[activeIdx] : null;
   278→
   279→    if (activeAcc) {
   280→      els.activeStatusText.textContent = `${activeAcc.name} (${toFa(activeIdx + 1)} از ${toFa(total)})`;
   281→      els.activeStatusText.style.color = '#34d399';
   282→    } else if (total > 0) {
   283→      els.activeStatusText.textContent = 'نشست جدید (ذخیره‌نشده)';
   284→      els.activeStatusText.style.color = '#fbbf24';
   285→    } else {
   286→      els.activeStatusText.textContent = 'بدون اکانت ذخیره‌شده';
   287→      els.activeStatusText.style.color = '#94a3b8';
   288→    }
   289→  }
   290→
   291→  const hasActive = state.accounts.some((a) => a.id === state.activeAccountId);
   292→  const canCycle = total > 1 || (total === 1 && !hasActive);
   293→  els.nextAccountBtn.disabled = !canCycle;
   294→  els.prevAccountBtn.disabled = !canCycle;
   295→  els.openBatchBtn.disabled = total === 0;
   296→
   297→  els.searchInput.parentElement.classList.toggle('hidden', total < 4);
   298→  els.emptyAccountsState.classList.toggle('hidden', total > 0);
   299→
   300→  renderAccountsList();
   301→}

   303→function renderAccountsList() {
   304→  els.accountsList.innerHTML = '';
   305→  const q = searchQuery.trim().toLowerCase();
   306→
   307→  const openTabByAccId = new Map();
   308→  for (const item of state.openIsolatedTabs) {
   309→    openTabByAccId.set(item.accountId, item.tabId);

   312→  state.accounts.forEach((acc, idx) => {
   313→    if (q) {
   314→      const hay = `${acc.name} ${acc.note || ''} ${acc.email || ''} ${idx + 1} ${toFa(idx + 1)}`.toLowerCase();
   315→      if (!hay.includes(q)) return;
   316→    }
   317→
   318→    const effectiveCurrentId = state.isIsolatedTab
   319→      ? state.isolatedAccountId
   320→      : state.activeAccountId;
   321→    const isActiveHere = acc.id === effectiveCurrentId;
   322→    const openIsoTabId = openTabByAccId.get(acc.id);

   325→    card.className =
   326→      'acc-card' +
   327→      (isActiveHere ? ' active' : '') +
   328→      (acc.skipped ? ' skipped' : '');
   329→
   330→    const main = document.createElement('div');
   331→    main.className = 'acc-main';
   332→    main.title = `کلیک برای سوییچ در همین تب به «${acc.name}»`;
   333→
   334→    const indexBadge = document.createElement('div');
   335→    indexBadge.className = 'acc-index';
   336→    indexBadge.style.background = acc.color || '#6366f1';
   337→    indexBadge.textContent = toFa(idx + 1);
   338→
   339→    const info = document.createElement('div');
   340→    info.className = 'acc-info';
   341→
   342→    const titleRow = document.createElement('div');
   343→    titleRow.className = 'acc-title-row';

   345→    const nameEl = document.createElement('span');
   346→    nameEl.className = 'acc-name';

   353→      activePill.textContent = state.isIsolatedTab ? 'این تب' : 'فعال';

   360→      isoPill.title = 'کلیک برای رفتن به تب بازِ این اکانت';

   372→      skipPill.className = 'pill-badge pill-skip';
   373→      skipPill.textContent = 'توقف';

   464→        type: 'EDIT_ACCOUNT_META',
   465→        siteKey: state.siteKey,
   466→        accountId: acc.id,
   467→        patch: { skipped: !acc.skipped }

   469→      await loadState();
   470→    });
   471→
   472→    // Edit
   473→    const editBtn = document.createElement('button');
   474→    editBtn.className = 'mini-action';
   475→    editBtn.title = 'ویرایش نام، یادداشت و رنگ';
   476→    editBtn.textContent = '✏️';
   477→    editBtn.addEventListener('click', (e) => {
   478→      e.stopPropagation();
   479→      openEditModal(acc);
   480→    });
   481→
   482→    // Delete
   483→    const delBtn = document.createElement('button');
   484→    const isConfirming = pendingDeleteId === acc.id;
   485→    delBtn.className = 'mini-action' + (isConfirming ? ' confirm-del' : '');
   486→    delBtn.title = 'حذف اکانت';
   487→    delBtn.textContent = isConfirming ? 'حذف؟' : '🗑️';
   488→    delBtn.addEventListener('click', async (e) => {
   489→      e.stopPropagation();
   490→      if (pendingDeleteId !== acc.id) {
   491→        pendingDeleteId = acc.id;
   492→        renderAccountsList();
   493→        setTimeout(() => {
   494→          if (pendingDeleteId === acc.id) {
   495→            pendingDeleteId = null;
   496→            renderAccountsList();
   497→          }
   498→        }, 3000);
   499→        return;
   500→      }
   501→      pendingDeleteId = null;
   502→      const res = await sendMsg({
   503→        type: 'DELETE_ACCOUNT',
   504→        siteKey: state.siteKey,
   505→        accountId: acc.id
   506→      });
   507→      if (res.ok) {
   508→        showToast(`«${acc.name}» حذف شد.`);
   509→        await loadState();
   510→      }
   511→    });
   512→
   513→    actions.appendChild(isoBtn);
   514→    actions.appendChild(updateBtn);
   515→    actions.appendChild(skipBtn);
   516→    actions.appendChild(editBtn);
   517→    actions.appendChild(delBtn);
   518→
   519→    card.appendChild(main);
   520→    card.appendChild(actions);
   521→    els.accountsList.appendChild(card);
   522→  });
   523→}

   744→function htmlToCleanText(htmlArr, fallbackText) {
   745→  if (fallbackText && fallbackText.trim()) return fallbackText.trim();
   746→  if (!Array.isArray(htmlArr) || !htmlArr.length) return '(پیام بدون متن)';
   747→  try {
   748→    const doc = new DOMParser().parseFromString(htmlArr.join('\n'), 'text/html');
   749→    doc.querySelectorAll('script, style, noscript, iframe, object').forEach((el) => el.remove());
   750→    return (doc.body?.innerText || doc.body?.textContent || '').trim() || '(پیام بدون متن)';
   751→  } catch {
   752→    return '(خطا در نمایش متن پیام)';
   753→  }
   754→}
