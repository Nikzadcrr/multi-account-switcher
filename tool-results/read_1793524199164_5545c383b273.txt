    66→async function copyText(text) {
    67→  try {
    68→    if (navigator.clipboard?.writeText) {
    69→      await navigator.clipboard.writeText(text);
    70→      return true;
    71→    }
    72→  } catch {}
    73→  try {
    74→    const ta = document.createElement('textarea');
    75→    ta.value = text;
    76→    ta.setAttribute('readonly', '');
    77→    ta.style.cssText = 'position:fixed;top:-999px;left:-999px;opacity:0;pointer-events:none;';
    78→    document.body.appendChild(ta);
    79→    ta.select();
    80→    ta.setSelectionRange(0, text.length);
    81→    const ok = document.execCommand('copy');
    82→    ta.remove();
    83→    if (ok) return true;
    84→  } catch {}
    85→  return false;
    86→}

    93→function formatRelativeTime(ts) {
    94→  if (!ts) return 'استفاده‌نشده';
    95→  const diffSec = Math.max(0, Math.floor((Date.now() - ts) / 1000));
    96→  if (diffSec < 45) return 'هم‌اکنون';
    97→  const mins = Math.floor(diffSec / 60);
    98→  if (mins < 60) return `${toFa(mins)} دقیقه پیش`;
    99→  const hours = Math.floor(mins / 60);
   100→  if (hours < 24) return `${toFa(hours)} ساعت پیش`;
   101→  const days = Math.floor(hours / 24);
   102→  return `${toFa(days)} روز پیش`;
   103→}

   105→function sendMsg(payload) {
   106→  return new Promise((resolve) => {
   107→    chrome.runtime.sendMessage(payload, (res) => {
   108→      if (chrome.runtime.lastError) {
   109→        resolve({ ok: false, error: chrome.runtime.lastError.message });
   110→        return;
   111→      }
   112→      resolve(res || { ok: false, error: 'پاسخی دریافت نشد.' });
   113→    });
   114→  });
   115→}

   288→
   289→  const [res, mRes] = await Promise.all([

   291→    sendMsg({ type: 'MAIL_GET_STATE' })

   294→  if (mRes?.ok) {
   295→    mailState.activeMailboxId = mRes.activeMailboxId;
   296→    mailState.mailboxes = mRes.mailboxes || [];
   297→    await loadMailUiPref();
   298→    renderMailSections();
   299→  }
   300→
   301→  if (!res.ok || !res.supported) {
   302→    state.supported = false;

   304→    els.domainBadge.textContent = 'صفحه داخلی مرورگر';

   311→
   312→  state = {

   314→    tabId: tab?.id || res.tabId,
   315→    url: tab?.url || res.url,
   316→    supported: true,
   317→    siteKey: res.siteKey,
   318→    hostname: res.hostname,
   319→    isGoogleUnified: Boolean(res.isGoogleUnified),

   627→async function ensureProviderDomains(provider) {
   628→  if (!providerDomainsCache[provider]) {
   629→    const res = await sendMsg({ type: 'MAIL_FETCH_DOMAINS', provider });
   630→    providerDomainsCache[provider] = {
   631→      domains: res.ok && Array.isArray(res.domains) ? res.domains : [],
   632→      popular: res.ok && Array.isArray(res.popular) ? res.popular : []
   633→    };
   634→  }
   635→  return providerDomainsCache[provider];
   636→}

   975→function htmlToCleanText(htmlArr, fallbackText) {
   976→  if (fallbackText && fallbackText.trim()) return fallbackText.trim();
   977→  if (!Array.isArray(htmlArr) || !htmlArr.length) return '(پیام بدون متن)';
   978→  try {
   979→    const doc = new DOMParser().parseFromString(htmlArr.join('\n'), 'text/html');
   980→    doc.querySelectorAll('script, style, noscript, iframe, object').forEach((el) => el.remove());
   981→    return (doc.body?.innerText || doc.body?.textContent || '').trim() || '(پیام بدون متن)';
   982→  } catch {
   983→    return '(خطا در نمایش متن پیام)';
   984→  }
   985→}
